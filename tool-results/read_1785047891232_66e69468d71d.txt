   192→    // ═══════════════════════════════════════════════════════════════════════
   193→    // CRON-COMPLETIONIST-65: 3D PathNavigation — the single biggest behavior fix
   194→    // ═══════════════════════════════════════════════════════════════════════
   195→    //
   196→    // PROBLEM: SpiritBeastEntity extends PathfinderMob, which uses
   197→    // GroundPathNavigation by default. GroundPathNavigation ONLY paths on
   198→    // the XZ plane at the entity's feet Y level. It ignores altitude entirely.
   199→    //
   200→    // CONSEQUENCE: Flying beasts (hawk, bat, qilin) with FlightGoal set
   201→    // noGravity=true and fly at Y=ground+15, but their MeleeAttackGoal
   202→    // and HuntGoal use GroundPathNavigation to path to targets ON THE GROUND.
   203→    // The entity tries to reach a ground-level path node while floating 15
   204→    // blocks up — resulting in the entity drifting sideways toward the target
   205→    // but never descending, or clipping through terrain when FlightGoal
   206→    // bulldozes with setDeltaMovement.
   207→    //
   208→    // Aquatic beasts (sea_serpent, soul_fish) also use GroundPathNavigation,
   209→    // so their HuntGoal paths them along the ocean floor instead of through
   210→    // the water column. A sea serpent chasing fish on the sea floor looks
   211→    // absurd — it should swim through 3D water space.
   212→    //
   213→    // FIX: Override createNavigation() to return:
   214→    //   - FlyPathNavigation for flyers (HAWK, BAT, QILIN) — this checks
   215→    //     canMoveTo in 3D, allowing paths that go over/around obstacles
   216→    //   - WaterBoundPathNavigation for aquatics (SEA_SERPENT, SOUL_FISH)
   217→    //     — this prefers water blocks and paths through water volumes
   218→    //   - GroundPathNavigation for ground beasts (existing, no change)
   219→    //
   220→    // FlyPathNavigation exists in MC 1.20.1 (used by Phantom, Ghast).
   221→    // WaterBoundPathNavigation exists in MC 1.20.1 (used by Dolphin, Fish).
   222→    // We simply USE them instead of reinventing the wheel.
   223→    //
   224→    // This is the most impactful single change because it makes ALL goals
   225→    // (not just FlightGoal) respect 3D space. MeleeAttackGoal, HuntGoal,
   226→    // PatrolGoal, FleeGoal — ALL now path correctly in 3D for flyers and
   227→    // aquatics.
   228→
   229→    @Override
   230→    protected PathNavigation createNavigation(Level level) {
   231→        // Navigation is selected based on the CURRENT beast type.
   232→        // This is called during super() constructor, before defineSynchedData(),
   233→        // so we cannot read DATA_BEAST_TYPE yet. Default to ground navigation
   234→        // here — it will be overridden by reassessNavigation() after type is set.
   235→        // (PathfinderMob constructor calls createNavigation before defineSynchedData)
   236→        return new GroundPathNavigation(this, level);
   237→    }
   238→
   239→    /**
   240→     * CRON-COMPLETIONIST-65: Replace the PathNavigation with the correct
   241→     * 3D variant based on beast type. Called from setBeastType() and
   242→     * defineSynchedData() after DATA_BEAST_TYPE is available.
   243→     *
   244→     * This is separate from reassessMoveControl() because the navigation
   245→     * is a different system from the move control. MoveControl handles
   246→     * the physics of movement (altitude maintenance, obstacle vaulting).
   247→     * Navigation handles the pathfinding (finding a route to the target).
   248→     */
   249→    private void reassessNavigation() {
   250→        BeastType type = getBeastType();
   251→        if (type.isFlyer()) {
   252→            this.navigation = new SpiritFlightPathNavigation(this, this.level());
   253→        } else if (type.isAquatic()) {
   254→            this.navigation = new WaterBoundPathNavigation(this, this.level());
   255→        }
   256→        // Ground beasts keep GroundPathNavigation from createNavigation()
   257→    }

   602→        if (feedGoal != null) feedGoal.tickHunger();
   603→        if (!this.level().isClientSide()) {
   604→            int ticksSinceGoal = this.tickCount - this.goalPoseTick;
   605→            boolean goalDrivingPose = ticksSinceGoal < 5;
   606→
   607→            if (!goalDrivingPose) {
   608→                if (this.isInWater()) {
   609→                    setSpiritPose(POSE_SWIMMING);
   610→                } else if (!this.onGround()) {
   611→                    setSpiritPose(POSE_FLYING);
   612→                } else if (this.getDeltaMovement().lengthSqr() > 0.01D) {
   613→                    setSpiritPose(POSE_STANDING);
   614→                }
   615→            }
   616→        }
