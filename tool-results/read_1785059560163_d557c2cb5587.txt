    18→/**
    19→ * BlockPlacementEngine — translates a {@link ResidenceManifest} into
    20→ * Minecraft blocks placed in the world.
    21→ *
    22→ * <p>Per Article XLVII: "The house is literally generated from the resident."
    23→ * This engine is the rendering layer: it reads the manifest (which was derived
    24→ * from a ResidentProfile) and places the correct blocks for each room.
    25→ *
    26→ * <p>This engine is HAND-AUTHORED. Each room type has an explicit block-layout
    27→ * method. There is NO procedural generation. The block palette for each room
    28→ * purpose is explicit in code. If you change the palette, ALL residences of
    29→ * that type change consistently — exactly as Article XLVII requires.
    30→ *
    31→ * <p>Per Article XXVI: this is NOT a new Engine in the infrastructure sense.
    32→ * It is a placement utility that reads the existing ResidenceManifest and places
    33→ * blocks. No new bus, no new subscriber, no new store.
    34→ *
    35→ * <h2>Block Palette (Wang Family Village — humble mortal construction)</h2>
    36→ * <ul>
    37→ *   <li>Walls: OAK_PLANKS</li>
    38→ *   <li>Corners/Posts: OAK_LOG</li>
    39→ *   <li>Floor: OAK_PLANKS (interior), COBBLESTONE (foundation/exterior)</li>
    40→ *   <li>Ceiling: OAK_PLANKS (solid) or AIR (courtyard/outdoor)</li>
    41→ *   <li>Roof: OAK_STAIRS (sloped) or OAK_SLAB (flat)</li>
    42→ *   <li>Door: OAK_DOOR (normal) or IRON_DOOR (defensive)</li>
    43→ *   <li>Fence: OAK_FENCE + OAK_FENCE_GATE</li>
    44→ * </ul>
    45→ *
    46→ * <h2>Room placement flow</h2>
    47→ * <pre>
    48→ *   ResidenceManifest → placeResidence(level, origin)
    49→ *     → for each room: placeRoomByPurpose(level, roomOrigin, roomSpec)
    50→ *       → specific method per RoomPurpose (placeEntry, placeBedroom, etc.)
    51→ *         → helper methods: placeWalls, placeFloor, placeCeiling, placeBox
    52→ * </pre>
    53→ */
    54→public final class BlockPlacementEngine {

   116→    /**
   117→     * Place an entire residence from its manifest.
   118→     * The origin is the northwest corner of the bounding box.
   119→     * Each room's position is computed from a fixed floor plan.
   120→     *
   121→     * <p><b>IMPORTANT:</b> This method clears the area first (fills with AIR),
   122→     * then places the structure. This is safe for first-time placement but
   123→     * destructive if called twice at the same location.
   124→     *
   125→     * @param level   the server level
   126→     * @param origin  the northwest corner of the residence footprint
   127→     * @param manifest the residence manifest to place
   128→     */
   129→    public static void placeResidence(ServerLevel level, BlockPos origin, ResidenceManifest manifest) {
   130→        Ergenverse.LOGGER.info("[BlockPlacementEngine] Placing residence '{}' at {} ({} rooms)",
   131→                manifest.residenceLabel(), origin, manifest.roomCount());
   132→
   133→        // Phase 1: Clear the area
   134→        BlockPos southEast = origin.offset(13, 5, 9);
   135→        fillBox(level, origin, southEast, AIR);
   136→
   137→        // Phase 2: Place rooms by purpose
   138→        for (RoomSpec room : manifest.rooms()) {
   139→            BlockPos roomOrigin = resolveRoomOrigin(room.purpose(), origin);
   140→            placeRoomByPurpose(level, roomOrigin, room, manifest);
   141→        }
   142→
   143→        // Phase 3: Place roof over enclosed rooms
   144→        placeRoof(level, origin);
   145→
   146→        Ergenverse.LOGGER.info("[BlockPlacementEngine] Residence '{}' placed successfully",
   147→                manifest.residenceLabel());
   148→    }

   154→    /**
   155→     * Hand-authored floor plan for Wang Family Village residences.
   156→     * Each room has a fixed position relative to the residence origin (NW corner).
   157→     *
   158→     * <pre>
   159→     *   N
   160→     *   [MeditRm][Bedroom ][Bedroom Roof/Observation]
   161→     *   [Hidden  ][Bedroom ]
   162→     *   [Workshop][Kitchen ][Entry  ]
   163→     *           [Kitchen ][Storage]
   164→     *           [        ][Courtyard  ]
   165→     *                    E→
   166→     * </pre>
   167→     *
   168→     * Coordinates are in (east, south) offsets from the origin.
   169→     */
   170→    private static BlockPos resolveRoomOrigin(RoomPurpose purpose, BlockPos residenceOrigin) {
   171→        return switch (purpose) {
   172→            case ENTRY           -> residenceOrigin.offset(8, 0, 3);
   173→            case BEDROOM         -> residenceOrigin.offset(5, 0, 0);
   174→            case KITCHEN         -> residenceOrigin.offset(8, 0, 0);
   175→            case STORAGE         -> residenceOrigin.offset(8, 0, 6);
   176→            case COURTYARD       -> residenceOrigin.offset(8, 0, 7);
   177→            case MEDITATION_ROOM -> residenceOrigin.offset(0, 0, 0);
   178→            case OBSERVATION_POST -> residenceOrigin.offset(5, 4, 0); // on top of bedroom
   179→            case HIDDEN_STASH    -> residenceOrigin.offset(0, -1, 0); // under meditation room
   180→            case WORKSHOP        -> residenceOrigin.offset(8, 0, 3);
   181→            default -> residenceOrigin;
   182→        };
   183→    }

   208→    private static void placeRoomByPurpose(ServerLevel level, BlockPos origin,
   209→                                            RoomSpec room, ResidenceManifest manifest) {
   210→        switch (room.purpose()) {
   211→            case ENTRY -> placeEntry(level, origin, room, manifest);
   212→            case BEDROOM -> placeBedroom(level, origin, room, manifest);
   213→            case KITCHEN -> placeKitchen(level, origin, room, manifest);
   214→            case STORAGE -> placeStorage(level, origin, room, manifest);
   215→            case COURTYARD -> placeCourtyard(level, origin, room);
   216→            case MEDITATION_ROOM -> placeMeditationRoom(level, origin, room, manifest);
   217→            case OBSERVATION_POST -> placeObservationPost(level, origin, room);
   218→            case HIDDEN_STASH -> placeHiddenStash(level, origin, room);
   219→            case WORKSHOP -> placeWorkshop(level, origin, room, manifest);
   220→            default -> placeGenericRoom(level, origin, room);
   221→        }
   222→    }

   228→    /** Entry: 3x3x3 room with wooden door facing east. */
   229→    private static void placeEntry(ServerLevel level, BlockPos origin, RoomSpec room,
   230→                                   ResidenceManifest manifest) {
   231→        // Floor (cobblestone — entry is a threshold)
   232→        placeFloor(level, origin, origin.offset(2, 0, 2), FLOOR_COBBLE);
   233→
   234→        // Walls: north, south, west (east is open for door)
   235→        placeWallNorth(level, origin, origin.offset(2, 2, 2));
   236→        placeWallSouth(level, origin, origin.offset(2, 2, 2));
   237→        placeWallWest(level, origin, origin.offset(2, 2, 2));
   238→
   239→        // East wall with door gap at y=0
   240→        setBlock(level, origin.offset(2, 0, 1), AIR);
   241→        setBlock(level, origin.offset(2, 1, 1), AIR);
   242→        // Door
   243→        if (hasTrait(manifest, "cautious") || hasNeed(manifest, NeedCategory.DEFENSE)) {
   244→            setBlock(level, origin.offset(2, 0, 1),
   245→                Blocks.IRON_DOOR.defaultBlockState()
   246→                            .setValue(DoorBlock.FACING, Direction.EAST)
   247→                            .setValue(DoorBlock.OPEN, Boolean.FALSE));
   248→        } else {
   249→            setBlock(level, origin.offset(2, 0, 1),
   250→                Blocks.OAK_DOOR.defaultBlockState()
   251→                            .setValue(DoorBlock.FACING, Direction.EAST)
   252→                            .setValue(DoorBlock.OPEN, Boolean.FALSE));
   253→        }
   254→        setBlock(level, origin.offset(2, 1, 1),
   255→                Blocks.OAK_DOOR.defaultBlockState()
   256→                        .setValue(DoorBlock.HALF, DoubleBlockHalf.UPPER)
   257→                        .setValue(DoorBlock.FACING, Direction.EAST));
   258→
   259→        // Ceiling
   260→        placeCeiling(level, origin, origin.offset(2, 3, 2), CEILING_PLANKS);
   261→
   262→        // Corner posts
   263→        setBlock(level, origin, CORNER_LOG);
   264→        setBlock(level, origin.offset(2, 0, 0), CORNER_LOG);
   265→        setBlock(level, origin.offset(0, 0, 2), CORNER_LOG);
   266→        setBlock(level, origin.offset(2, 0, 2), CORNER_LOG);
   267→
   268→        // Torch for entry
   269→        setBlock(level, origin.offset(1, 2, 0), TORCH_WALL);
   270→
   271→        Ergenverse.LOGGER.debug("[BlockPlacementEngine]   Entry placed at {}", origin);
   272→    }

   274→    /** Bedroom: 5x3x4 room with bed, chest, and optional weapon. */
   275→    private static void placeBedroom(ServerLevel level, BlockPos origin, RoomSpec room,
   276→                                      ResidenceManifest manifest) {
   277→        BlockPos far = origin.offset(4, 2, 3);
   278→
   279→        // Floor
   280→        placeFloor(level, origin, far, FLOOR_PLANKS);
   281→
   282→        // Walls: all four sides (door gap in east wall at z=2)
   283→        placeWallsEnclosed(level, origin, far);
   284→        // Door gap in east wall
   285→        setBlock(level, far.east(), AIR);
   286→        setBlock(level, far.east().above(), AIR);
   287→
   288→        // Ceiling
   289→        placeCeiling(level, origin, far, CEILING_PLANKS);
   290→
   291→        // Corner posts
   292→        setBlock(level, origin, CORNER_LOG);
   293→        setBlock(level, far.offset(4, 0, 0), CORNER_LOG);
   294→        setBlock(level, origin.offset(0, 0, 3), CORNER_LOG);
   295→        setBlock(level, far.offset(4, 0, 3), CORNER_LOG);
   296→
   297→        // Bed: head at north wall (z=0), red wool, foot white wool
   298→        BlockPos bedHead = origin.offset(1, 1, 0);
   299→        setBlock(level, bedHead, BED_HEAD);
   300→        setBlock(level, bedHead.south(), BED_FOOT);
   301→
   302→        // Chest for personal items
   303→        setBlock(level, origin.offset(3, 1, 0), CHEST);
   304→
   305→        // Weapon near bed if cautious/concealment_first (Article XLVII §4)
   306→        if (hasTrait(manifest, "cautious") || hasTrait(manifest, "concealment_first")) {
   307→            setBlock(level, origin.offset(2, 1, 0), Blocks.IRON_BARS.defaultBlockState());
   308→        }
   309→
   310→        // Torches
   311→        setBlock(level, origin.offset(0, 2, 1), TORCH_WALL);
   312→        setBlock(level, origin.offset(0, 2, 2), TORCH_WALL);
   313→
   314→        Ergenverse.LOGGER.debug("[BlockPlacementEngine]   Bedroom placed at {}", origin);
   315→    }

   574→                // Sloped edge on north
   575→                if (z == 0) setBlock(level, pos.north(), ROOF_STAIRS_SOUTH);
   576→                // Sloped edge on south
   577→                if (z == 3) setBlock(level, pos.south(), ROOF_STAIRS_NORTH);

   686→    /**
   687→     * Convenience: set a single block.
   688→     *
   689→     * <p><b>CRON-COMPLETIONIST-61 — simulation provenance wiring:</b>
   690→     * When the {@link WorldRuntime} is initialized AND the target level is
   691→     * the Planet Suzaku level bound to the runtime, this routes the change
   692→     * through {@code WorldRuntime.get().world().setSimulationBlock(...)} so
   693→     * it is journaled under {@link dev.ergenverse.runtime.Provenance#SIMULATION}
   694→     * and persists across save/load via {@link dev.ergenverse.runtime.persist.WorldDeltaSavedData}.
   695→     *
   696→     * <p>Otherwise (non-Suzaku level, or runtime not yet initialized), it
   697→     * falls back to direct {@code level.setBlock}. This preserves the
   698→     * engine's behavior for ad-hoc uses outside the simulation (e.g. test
   699→     * worlds) while ensuring the canonical Suzaku simulation writes flow
   700→     * through the journal.
   701→     *
   702→     * <p>Architectural rationale: residences are SIMULATION, not CANON.
   703→     * They are placed in response to a resident's existence (Article XLVII:
   704→     * "the house is literally generated from the resident"), and a resident
   705→     * is a simulation actor. The blueprint does not contain per-block
   706→     * residence layouts — those are derived. So a residence's blocks belong
   707→     * in the SIMULATION layer, where they can diverge from a fresh save
   708→     * (e.g. a burned-down house stays burned-down across reload, but a new
   709→     * save starts fresh).
   710→     */
   711→    private static void setBlock(ServerLevel level, BlockPos pos, BlockState state) {
   712→        try {
   713→            WorldRuntime rt = WorldRuntime.get();
   714→            if (rt.isInitialized() && rt.suzakuLevel() == level) {
   715→                ResourceLocation rl = ForgeRegistries.BLOCKS.getKey(state.getBlock());
   716→                if (rl != null) {
   717→                    rt.world().setSimulationBlock(
   718→                            pos.getX(), pos.getY(), pos.getZ(), rl.toString());
   719→                    return;
   720→                }
   721→            }
   722→        } catch (Throwable t) {
   723→            Ergenverse.LOGGER.debug("[Ergenverse] BlockPlacementEngine facade write failed at {}: {}",
   724→                    pos, t.getMessage());
   725→        }
   726→        // Fallback: direct write (non-Suzaku level or runtime not initialized)
   727→        level.setBlock(pos, state, 3);
   728→    }
