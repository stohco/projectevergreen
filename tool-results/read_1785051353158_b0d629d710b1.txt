    28→/**
    29→ * WangLinItems — the DeferredRegister for ALL of Wang Lin's arsenal.
    30→ *
    31→ * <p>This is the single most important class for the "completionist Wang Lin branch"
    32→ * directive: it registers EVERY item, technique, pet, clone, and companion Wang Lin
    33→ * encountered or owned across the full Renegade Immortal narrative as an actual
    34→ * pickable Minecraft {@link Item}.
    35→ *
    36→ * <h2>Data-Driven Registration</h2>
    37→ * <p>The item list is loaded from {@code assets/ergenverse/wanglin_arsenal_manifest.json}
    38→ * (extracted from the Next.js prototype's {@code protagonist-arsenals.ts}). This manifest
    39→ * contains 309 entries — every Wang Lin arsenal asset with its canonical itemId,
    40→ * unlock threshold, and canon flag. The manifest is the single source of truth for
    41→ * what gets registered; adding a new Wang Lin item means adding it to the manifest,
    42→ * not hand-editing this class.
    43→ *
    44→ * <h2>Registration</h2>
    45→ * <p>Call {@link #register(IEventBus)} once during mod construction (from
    46→ * {@link dev.ergenverse.core.Ergenverse#Ergenverse}). This wires the DeferredRegister
    47→ * to the mod event bus. Forge then fires the RegisterEvent and all 309 items become
    48→ * real, holdable, creative-tab-listed Minecraft items.
    49→ *
    50→ * <h2>Access Pattern</h2>
    51→ * <p>After registration, look up any item by its canonId:
    52→ * <pre>{@code
    53→ * RegistryObject<Item> flag = WangLinItems.get("wanglin/restriction_flag");
    54→ * ItemStack stack = new ItemStack(flag.get());
    55→ * }</pre>
    56→ *
    57→ * <h2>Prime Directive Compliance</h2>
    58→ * <p>Every registered item is an objective world object — it exists regardless of
    59→ * player perception. The item's tooltip always shows its full canon info; perception
    60→ * gating (what the player UNDERSTANDS about the item) is handled by the
    61→ * {@link dev.ergenverse.perception.PerceptionEngine}, not by hiding the item.
    62→ */

   444→    /** A single entry from the arsenal manifest. */
   445→    public record ManifestEntry(String itemId, boolean isCanonical, int count, float unlockThreshold) {
   446→        /** The canonId portion (e.g. "wanglin/restriction_flag" from "ergen:wanglin/restriction_flag"). */
   447→        public String canonId() {
   448→            return itemId.replace("ergen:", "");
   449→        }
   450→        /** The registry name portion (e.g. "restriction_flag"). */
   451→        public String registryName() {
   452→            String id = itemId.replace("ergen:wanglin/", "");
   453→            return id;
   454→        }
   455→    }

   457→    /**
   458→     * Load the arsenal manifest from the embedded JSON resource and register every
   459→     * entry as a {@link WangLinItem}. This runs once at class-init (before the mod
   460→     * event bus fires the RegisterEvent).
   461→     */
   462→    private static synchronized void loadManifestAndRegister() {
   463→        if (registered) return;
   464→        Ergenverse.LOGGER.info("[WangLin] Loading arsenal manifest from assets/ergenverse/wanglin_arsenal_manifest.json...");
   465→
   466→        try (var is = WangLinItems.class.getResourceAsStream("/assets/ergenverse/wanglin_arsenal_manifest.json")) {
   467→            if (is == null) {
   468→                Ergenverse.LOGGER.error("[WangLin] ARSENAL MANIFEST NOT FOUND! No Wang Lin items will be registered.");
   469→                Ergenverse.LOGGER.error("[WangLin] Expected: src/main/resources/assets/ergenverse/wanglin_arsenal_manifest.json");
   470→                registered = true;
   471→                return;
   472→            }
   473→            var reader = new InputStreamReader(is, StandardCharsets.UTF_8);
   474→            JsonObject root = new Gson().fromJson(reader, JsonObject.class);
   475→            JsonArray entries = root.getAsJsonArray("entries");
   476→            int count = 0;
   477→            for (JsonElement el : entries) {
   478→                JsonObject e = el.getAsJsonObject();
   479→                String itemId = e.get("itemId").getAsString();
   480→                boolean isCanonical = e.get("isCanonical").getAsBoolean();
   481→                int cnt = e.has("count") ? e.get("count").getAsInt() : 1;
   482→                float unlock = e.has("unlockThreshold") ? e.get("unlockThreshold").getAsFloat() : 0f;
   483→
   484→                ManifestEntry me = new ManifestEntry(itemId, isCanonical, cnt, unlock);
   485→                MANIFEST.add(me);
   486→                registerArsenalItem(me);
   487→                count++;
   488→            }
   489→            Ergenverse.LOGGER.info("[WangLin] Arsenal manifest loaded: {} items registered.", count);
   490→        } catch (Exception e) {
   491→            Ergenverse.LOGGER.error("[WangLin] Failed to load arsenal manifest!", e);
   492→        }
   493→        registered = true;
   494→    }

   496→    /**
   497→     * Register a single arsenal item. The item is created as a {@link WangLinItem}
   498→     * with category inferred from the canonId prefix and evolution chain looked up
   499→     * from {@link ItemEvolutionRegistry}.
   500→     *
   501→     * <p>The {@code canonId} passed to {@link WangLinItem} is the <b>canonical
   502→     * registry entry id</b> (looked up from {@link #MANIFEST_TO_REGISTRY}), NOT
   503→     * the manifest's {@code itemId}. This ensures {@link WangLinItem}'s tooltip
   504→     * can call {@link WangLinMasterRegistry#lookup(String)} and find a match.
   505→     * Items whose manifest bare-id has no confident registry match get a
   506→     * {@code null} canonId — their tooltip enrichment is silently skipped
   507→     * (no fabricated matches).
   508→     */
   509→    private static void registerArsenalItem(ManifestEntry me) {
   510→        String registryName = me.registryName();
   511→        String manifestCanonId = me.canonId();
   512→
   513→        // CRON-SMOKE-TEST FIX: Skip items whose registry name is ALREADY owned by
   514→        // ErgenverseItems. Both DeferredRegisters target the "ergenverse" namespace,
   515→        // so if ErgenverseItems registers "ergenverse:soul_gourd" (the real
   516→        // SoulGourdItem with capture/release mechanics) and WangLinItems ALSO
   517→        // registers "ergenverse:soul_gourd" (a tooltip-only WangLinItem), Forge
   518→        // creates an override that maps to `air`, crashing the server at registry
   519→        // freeze with "One or more entry values did not copy to the correct id."
   520→        // The ErgenverseItems version is always the one with real mechanics, so we
   521→        // defer to it and skip the arsenal's display-only duplicate.
   522→        if (ERGENVERSE_ITEMS_OWNED_NAMES.contains(registryName)) {
   523→            Ergenverse.LOGGER.info("[WangLin] Skipping arsenal duplicate of ErgenverseItems-owned item: {}", registryName);
   524→            return;
   525→        }
   526→
   527→        // Skip duplicate entries — the manifest has 6 duplicate registry names.
   528→        // Without this check, DeferredRegister.register() throws IllegalArgumentException
   529→        // on the 2nd occurrence, aborting the loop and leaving all subsequent items unregistered.
   530→        if (!REGISTERED_NAMES.add(registryName)) {
   531→            Ergenverse.LOGGER.warn("[WangLin] Skipping duplicate arsenal item: {}", registryName);
   532→            return;
   533→        }
   534→
   535→        // Look up the actual canonical registry entry id from the manifest->registry map.
   536→        // If no confident match, leave canonId null (tooltip enrichment silently skipped).
   537→        String registryCanonId = MANIFEST_TO_REGISTRY.get(registryName);
   538→        if (registryCanonId == null) {
   539→            Ergenverse.LOGGER.debug("[WangLin] No registry match for arsenal item '{}' — tooltip enrichment will be skipped.",
   540→                    registryName);
   541→        }
   542→        WangLinItem.ArsenalCategory category = inferCategory(manifestCanonId);
   543→        String displayName = humanizeName(registryName);
   544→
   545→        // CANON-ONLY: We do NOT create default chains for items without explicit
   546→        // canon histories. Per the rewritten Prime Directive, items with a single
   547→        // documented canon form exist in that form — period. No invented "Peak"
   548→        // or "Awakened" stages. The ItemEvolutionRegistry contains ONLY chains
   549→        // for items with explicitly documented canon transformations.
   550→        ItemEvolutionRegistry.bootstrap();
   551→
   552→        // SPECIAL CASES: Items that need their own Item subclass with real mechanics.
   553→        // These are canon-critical items where WangLinItem's tooltip-only behavior
   554→        // is insufficient — the player must be able to LAUNCH a flying sword,
   555→        // STORE items in a ring, or CAPTURE souls with the gourd.
   556→        //
   557→        // CRON-COMPLETIONIST-67: Wired 4 flying swords from WangLinItem (display-only)
   558→        // to FlyingSwordItem (launchable projectile with cultivation-scaled damage
   559→        // and per-sword supernatural effects).
   560→        final RegistryObject<Item> ro;
   561→        if ("heaven_defying_bead".equals(registryName)) {
   562→            ro = ITEMS.register(registryName, () -> new HeavenDefyingBeadItem(
   563→                    new Item.Properties()));
   564→        } else if ("wealth_flying_sword".equals(registryName)) {
   565→            // Wang Lin's FIRST flying sword — the iconic Wealth Sword.
   566→            // Canon: basic qi blade, no special effect, but reliable.
   567→            // Damage: 8.0 (low — it's his weakest sword). Effect: NONE.
   568→            ro = ITEMS.register(registryName, () -> new FlyingSwordItem(
   569→                    8.0F, SwordEffectType.NONE,
   570→                    new Item.Properties().rarity(net.minecraft.world.item.Rarity.RARE)
   571→            ));
   572→        } else if ("core_treasure_sword".equals(registryName)) {
   573→            // Core Treasure Sword — can teleport targets on hit.
   574→            // Canon: a refined treasure-tier sword that displaces enemies.
   575→            // Damage: 12.0 (mid-tier). Effect: TELEPORT.
   576→            ro = ITEMS.register(registryName, () -> new FlyingSwordItem(
   577→                    12.0F, SwordEffectType.TELEPORT,
   578→                    new Item.Properties().rarity(net.minecraft.world.item.Rarity.RARE)
   579→            ));
   580→        } else if ("dark_green_flying_sword".equals(registryName)) {
   581→            // Dark Green Flying Sword — seeps poison into wounds.
   582→            // Canon: a sinister sword that poisons those it cuts.
   583→            // Damage: 10.0 (mid-tier). Effect: POISON (Wither II, 3s).
   584→            ro = ITEMS.register(registryName, () -> new FlyingSwordItem(
   585→                    10.0F, SwordEffectType.POISON,
   586→                    new Item.Properties().rarity(net.minecraft.world.item.Rarity.RARE)
   587→            ));
   588→        } else if ("blood_slaughter_sword".equals(registryName)) {
   589→            // Blood Slaughter Sword — drains life force from those it cuts.
   590→            // Canon: one of the Seven Swords of the Ancient Dao, absorbs blood qi.
   591→            // Damage: 15.0 (high-tier). Effect: LIFESTEAL (30% damage healed).
   592→            ro = ITEMS.register(registryName, () -> new FlyingSwordItem(
   593→                    15.0F, SwordEffectType.LIFESTEAL,
   594→                    new Item.Properties().rarity(net.minecraft.world.item.Rarity.EPIC)
   595→            ));
   596→        } else {
   597→            ro = ITEMS.register(registryName, () -> new WangLinItem(
   598→                    new Item.Properties(),
   599→                    registryCanonId,
   600→                    displayName,
   601→                    "",  // CN name filled in by evolution-chain stages if available
   602→                    category,
   603→                    me.isCanonical() ? 5 : 3,
   604→                    manifestCanonId,
   605→                    me.unlockThreshold(),
   606→                    "Renegade Immortal \u2014 Wang Lin's arsenal"
   607→            ));
   608→        }
   609→        ARSENAL.put(manifestCanonId, ro);
   610→    }

   612→    /** Infer the arsenal category from the canonId naming conventions. */
   613→    private static WangLinItem.ArsenalCategory inferCategory(String canonId) {
   614→        if (canonId.contains("essence")) return WangLinItem.ArsenalCategory.ESSENCE;
   615→        if (canonId.contains("domain")) return WangLinItem.ArsenalCategory.DOMAIN;
   616→        if (canonId.contains("bridge")) return WangLinItem.ArsenalCategory.DOMAIN;
   617→        if (canonId.contains("pet") || canonId.contains("dragon") || canonId.contains("soul_tornado")
   618→                || canonId.contains("monkey") || canonId.contains("beast")) return WangLinItem.ArsenalCategory.PET;
   619→        if (canonId.contains("devil") || canonId.contains("clone") || canonId.contains("xu_liguo")) return WangLinItem.ArsenalCategory.CLONE;
   620→        if (canonId.contains("companion")) return WangLinItem.ArsenalCategory.COMPANION;
   621→        if (canonId.contains("formation") || canonId.contains("array")) return WangLinItem.ArsenalCategory.FORMATION;
   622→        if (canonId.contains("flag") || canonId.contains("sword") || canonId.contains("blade")
   623→                || canonId.contains("chariot") || canonId.contains("crown") || canonId.contains("hat")
   624→                || canonId.contains("pen") || canonId.contains("coffin") || canonId.contains("compass")
   625→                || canonId.contains("net") || canonId.contains("shield") || canonId.contains("screen")
   626→                || canonId.contains("pavilion")) return WangLinItem.ArsenalCategory.ARTIFACT;
   627→        if (canonId.contains("art") || canonId.contains("technique") || canonId.contains("spell")
   628→                || canonId.contains("hex") || canonId.contains("print") || canonId.contains("sound")
   629→                || canonId.contains("method") || canonId.contains("refining") || canonId.contains("detection")
   630→                || canonId.contains("escape") || canonId.contains("transformation") || canonId.contains("searching")
   631→                || canonId.contains("rotation") || canonId.contains("bending") || canonId.contains("tactic")
   632→                || canonId.contains("extraction") || canonId.contains("devouring") || canonId.contains("disguising")
   633→                || canonId.contains("erasing") || canonId.contains("stealing") || canonId.contains("ascension")) return WangLinItem.ArsenalCategory.TECHNIQUE;
   634→        return WangLinItem.ArsenalCategory.MISCELLANEOUS;
   635→    }

   637→    /** Convert a registry name like "restriction_flag" to "Restriction Flag". */
   638→    private static String humanizeName(String registryName) {
   639→        StringBuilder sb = new StringBuilder();
   640→        for (String word : registryName.split("_")) {
   641→            if (!word.isEmpty()) {
   642→                if (sb.length() > 0) sb.append(' ');
   643→                sb.append(Character.toUpperCase(word.charAt(0))).append(word.substring(1));
   644→            }
   645→        }
   646→        return sb.toString();
   647→    }
