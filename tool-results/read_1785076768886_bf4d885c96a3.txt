    24→/**
    25→ * EntityCultivator — a polymorphic NPC shell driven by canon data.
    26→ *
    27→ * <h2>Design</h2>
    28→ * <p>Instead of 151 separate Java classes for 151 canon characters, this is
    29→ * a single, highly polymorphic entity. At spawn, it reads a
    30→ * {@code character_id} (e.g. {@code "wang_tiangui"}) and configures itself
    31→ * from the canon database + runtime overrides:
    32→ * <ol>
    33→ *   <li>{@link WorldStateDataLoader#getEntry(String, String)} — reads the
    34→ *       canon baseline (name, cultivation realm, HP, location, etc.) from
    35→ *       {@code data/ergenverse/npcs/<id>.json}.</li>
    36→ *   <li>{@link WorldRuntimeState#getNpcState(String)} — layers any t>0
    37→ *       mutations (damage taken, position drift) on top of the baseline.</li>
    38→ * </ol>
    39→ *
    40→ * <h2>Synced data</h2>
    41→ * <p>{@link #DATA_CHARACTER_ID} is a {@link SynchedEntityData} string field,
    42→ * synced to clients. The client renderer reads it to pick the correct
    43→ * model/texture. Without this, every cultivator would render identically.
    44→ *
    45→ * <h2>Hibernation</h2>
    46→ * <p>{@link MobCategory#MISC} entities normally despawn when players walk
    47→ * away. We override {@link #removeWhenFarAway(double)} to return
    48→ * {@code false} — canon entities never despawn. But to avoid ticking-entity
    49→ * accumulation (100s of NPCs each running AI), {@link #aiStep()} short-circuits
    50→ * when no player is within {@link #HIBERNATION_RANGE} blocks. The entity
    51→ * persists in the world but consumes minimal CPU.
    52→ *
    53→ * <h2>State persistence</h2>
    54→ * <p>On chunk unload ({@link #onRemovedFromLevel()}), the entity writes its
    55→ * current state (HP, position) back to {@link WorldRuntimeState}. On the
    56→ * next materialization, {@link #initializeFromData} re-reads it. The canon
    57→ * DB is never written to.
    58→ *
    59→ * <h2>v1 scope</h2>
    60→ * <p>This first version supports {@code wang_tiangui} (Wang Lin's mortal
    61→ * father): Mortal realm, 20 HP, no combat, passive wander. Behavior AI
    62→ * (combat, dialogue, sect routines) is deferred to v2. The shell is
    63→ * complete; the behavior brain is not.
    64→ *
    65→ * <h2>Prime Directive</h2>
    66→ * <p>"Reality is objective." The entity's stats are read from canon, not
    67→ * invented. If canon is silent on a stat, the field is left at its Minecraft
    68→ * default (marked with a TODO) rather than fabricated.
    69→ */

   348→    // ── Activity lock (reasoning-engine cycle) ──────────────────────────

   350→    /**
   351→     * When non-zero, the entity is <b>activity-locked</b> — its AI is suppressed
   352→     * and it holds its current position + pose. This is how the
   353→     * {@link dev.ergenverse.simulation.settlement.ActorReasoningEngine} makes
   354→     * an entity visibly BEHAVE its reasoning: Wang Lin freezes at the treeline
   355→     * in POSE_OBSERVING, the patriarch freezes at the gate in POSE_GUARDING,
   356→     * while the others flee home. The lock expires at this tick; the
   357→     * {@link dev.ergenverse.simulation.settlement.ActorMaterializer} clears it
   358→     * when the threat expires, resuming normal AI.
   359→     *
   360→     * <p>Per the user's directive: the same wolf event should produce visibly
   361→     * different behavior. Without this lock, the entity's RandomStrollGoal
   362→     * would wander Wang Lin away from his observing vantage immediately. The
   363→     * lock makes the reasoning-derived position + pose <b>observable</b>.
   364→     */
   365→    private long activityLockExpiryTick = 0L;
   366→
   367→    /** True if the entity is currently activity-locked (AI suppressed, holding pose). */
   368→    public boolean isActivityLocked() {
   369→        return activityLockExpiryTick > 0L;
   370→    }
   371→
   372→    /**
   373→     * Lock the entity to a reasoning-derived activity: suppress AI, set the
   374→     * pose, hold position until the expiry tick.
   375→     */
   376→    public void lockToActivity(int pose, long expiryTick) {
   377→        this.entityData.set(DATA_POSE, pose);
   378→        this.activityLockExpiryTick = expiryTick;
   379→        // AI suppression happens in {@link #aiStep()} (checking
   380→        // isActivityLocked). We deliberately do NOT use setNoAi(true) because
   381→        // that flag persists to NBT — if the chunk unloaded while locked, the
   382→        // entity would be stuck frozen on reload. The transient field approach
   383→        // means the lock clears on reload (the threat will have expired or the
   384→        // materializer will re-evaluate).
   385→    }
   386→
   387→    /**
   388→     * Release the activity lock: resume normal AI. Called by the
   389→     * {@link dev.ergenverse.simulation.settlement.ActorMaterializer} when the
   390→     * threat has expired and the actor should return to daily rhythm.
   391→     */
   392→    public void releaseActivityLock() {
   393→        if (activityLockExpiryTick > 0L) {
   394→            this.activityLockExpiryTick = 0L;
   395→            this.entityData.set(DATA_POSE, POSE_IDLE);
   396→        }
   397→    }

   668→    /**
   669→     * Apply canon baseline fields from the NPC JSON.
   670→     * Reads: name_cn, name_en, cultivation_realm, max_hp (if canon).
   671→     */
   672→    private void applyCanonBaseline(JsonObject data) {
   673→        // Display name — prefer name_cn, fall back to name_en, then ID
   674→        String nameCn = hasString(data, "name_cn") ? data.get("name_cn").getAsString() : null;
   675→        String nameEn = hasString(data, "name_en") ? data.get("name_en").getAsString() : null;
   676→        String displayName = nameCn != null ? nameCn : (nameEn != null ? nameEn : "Unknown");
   677→        if (nameCn != null && nameEn != null) {
   678→            displayName = nameCn + " (" + nameEn + ")";
   679→        }
   680→        this.setDisplayNameCn(displayName);
   681→
   682→        // Cultivation realm
   683→        if (hasString(data, "cultivation_realm")) {
   684→            this.setCultivationRealm(data.get("cultivation_realm").getAsString());
   685→        } else if (hasString(data, "realm")) {
   686→            this.setCultivationRealm(data.get("realm").getAsString());
   687→        }
   688→
   689→        // Max HP — only if canon specifies (don't invent)
   690→        if (hasNumber(data, "max_hp")) {
   691→            float maxHp = data.get("max_hp").getAsFloat();
   692→            this.getAttribute(Attributes.MAX_HEALTH).setBaseValue(maxHp);
   693→            this.setHealth(maxHp);
   694→        }
   695→
   696→        // CRON-COMPLETIONIST-50: Sect/faction — read from "sect", "faction", or "affiliation"
   697→        String sectId = null;
   698→        if (hasString(data, "sect")) {
   699→            sectId = data.get("sect").getAsString();
   700→        } else if (hasString(data, "faction")) {
   701→            sectId = data.get("faction").getAsString();
   702→        } else if (hasString(data, "affiliation")) {
   703→            sectId = data.get("affiliation").getAsString();
   704→        }
   705→        // Normalize: lowercase, spaces to underscores, strip parentheticals
   706→        if (sectId != null && !sectId.isEmpty()) {
   707→            sectId = sectId.toLowerCase().replaceAll("\\s+", "_")
   708→                    .replaceAll("[()]", "").replaceAll("_+", "_");
   709→            // Strip trailing underscore if any
   710→            if (sectId.endsWith("_")) sectId = sectId.substring(0, sectId.length() - 1);
   711→            this.setSectId(sectId);
   712→        }
   713→        // v1: mortal default (20 HP) applies via createAttributes() if canon silent
   714→    }

   837→    /**
   838→     * Hibernation: if no player is within {@link #HIBERNATION_RANGE}, skip
   839→     * AI processing. The entity still exists in the world (rendered if in
   840→     * view, loaded in chunk), but its brain is off — no pathfinding, no
   841→     * goal ticking. This prevents the "100 NPCs all ticking" performance
   842→     * death trap.
   843→     */
   844→    @Override
   845→    public void aiStep() {
   846→        if (this.level().isClientSide) {
   847→            // Client: always animate (rendering layer handles LOD)
   848→            super.aiStep();
   849→            return;
   850→        }
   851→        // Activity lock: the reasoning engine has placed this entity at a
   852→        // specific position with a specific pose (Wang Lin observing at the
   853→        // treeline, the patriarch guarding the gate). Suppress all goal AI so
   854→        // the entity holds its reasoning-derived state and the player can SEE
   855→        // the differentiated behavior. Gravity/drowning still apply via super.
   856→        if (isActivityLocked()) {
   857→            this.goalSelector.getRunningGoals().forEach(g -> g.stop());
   858→            super.aiStep();
   859→            return;
   860→        }
   861→        // CRON-COMPLETIONIST-19: Cognitive attention lock. When the NPC is
   862→        // cognition-absorbed (a Commitment is active and the CognitionDrivenGoal
   863→        // has set the attention lock), suppress RandomLookAroundGoal so the
   864→        // NPC's head tracks the cognitive look-target, not random stuff. The
   865→        // user's directive: an observing NPC "doesn't respond immediately to
   866→        // player." We do NOT stop ALL goals — CognitionDrivenGoal must keep
   867→        // running (it owns the look-target). We only stop the random-look
   868→        // specifically. Other goals (combat, gift-offer) still fire if their
   869→        // conditions trigger, which is correct — an observing NPC attacked
   870→        // by a zombie should still defend itself.
   871→        if (cognitiveAttentionLock) {
   872→            this.goalSelector.getRunningGoals()
   873→                    .filter(wg -> wg.getGoal() instanceof net.minecraft.world.entity.ai.goal.RandomLookAroundGoal)
   874→                    .forEach(wg -> wg.stop());
   875→        }
   876→        // Server: hibernate if no player nearby
   877→        if (this.level().getNearestPlayer(this, HIBERNATION_RANGE) == null) {
   878→            // Hibernating — minimal tick. Still apply gravity, drowning, etc.
   879→            // but skip goalSelector and navigation updates.
   880→            this.goalSelector.getRunningGoals().forEach(g -> g.stop());
   881→            // Call super but with navigation disabled — gravity still works
   882→            super.aiStep();
   883→            return;
   884→        }
   885→        super.aiStep();

   888→    /**
   889→     * Sync current entity state to the runtime layer.
   890→     *
   891→     * <p>Called from {@link #addAdditionalSaveData} (which fires on chunk
   892→     * unload — Minecraft serializes the entity to NBT when its chunk
   893→     * unloads) and from {@link #die} (on death). This is the dematerialization
   894→     * step: the entity's current HP and position are written to
   895→     * {@link WorldRuntimeState} so the next materialization reads the
   896→     * updated state.
   897→     *
   898→     * <p>Note: in MC 1.20.1, {@code Entity.setRemoved(RemovalReason)} is final
   899→     * and cannot be overridden. The NBT save path is the correct hook for
   900→     * chunk-unload persistence — it fires at the same time and is the
   901→     * vanilla-sanctioned way to persist entity state.
   902→     */
   903→    private void syncStateToRuntime() {
   904→        if (this.level().isClientSide || !this.initialized || this.getCharacterId().isEmpty()) return;
   905→        ServerLevel serverLevel = (ServerLevel) this.level();
   906→        WorldRuntimeState runtime = WorldRuntimeState.get(serverLevel);
   907→        CompoundTag state = runtime.getNpcState(this.getCharacterId());
   908→        if (state == null) {
   909→            state = new CompoundTag();
   910→        }
   911→        // Record current HP (the most common mutation: damage taken)
   912→        state.putFloat("current_hp", this.getHealth());
   913→        // Record last position (for re-materialization at the same spot)
   914→        state.putDouble("last_pos_x", this.getX());
   915→        state.putDouble("last_pos_y", this.getY());
   916→        state.putDouble("last_pos_z", this.getZ());
   917→        state.putLong("last_seen_tick", serverLevel.getGameTime());
   918→        runtime.updateNpcState(this.getCharacterId(), state);
   919→    }

   927→        super.addAdditionalSaveData(compound);
   928→        compound.putString("CharacterId", this.getCharacterId());
   929→        compound.putString("DisplayName", this.getDisplayNameCn());
   930→        compound.putString("CultivationRealm", this.getCultivationRealm());
   931→        compound.putBoolean("Initialized", this.initialized);
   932→        compound.putInt("CultivatorPose", this.getCultivatorPose());
   933→        // Sync to runtime layer on every NBT save (fires on chunk unload).
   934→        // This is the dematerialization persistence hook.

   936→        // Sever the ActorEntityLink — the entity is being serialized/unloaded.
   937→        // The Actor remains in the registry (simulation continues at territory
   938→        // level), but the link to this Minecraft entity is gone.
   939→        if (!this.level().isClientSide) {
   940→            dev.ergenverse.simulation.intent.ActorEntityLink.onEntityUnload(this);
   941→        }
   942→    }

   947→        this.setCharacterId(compound.getString("CharacterId"));
   948→        if (compound.contains("DisplayName")) {
   949→            this.setDisplayNameCn(compound.getString("DisplayName"));
   950→        }
   951→        if (compound.contains("CultivationRealm")) {
   952→            this.setCultivationRealm(compound.getString("CultivationRealm"));
   953→        }
   954→        this.initialized = compound.getBoolean("Initialized");
   955→        if (compound.contains("CultivatorPose")) {
   956→            this.setCultivatorPose(compound.getInt("CultivatorPose"));
   957→        }
   958→        // Re-establish the ActorEntityLink on chunk reload.
   959→        // The entity is materializing from NBT — link it back to its Actor.
   960→        if (!this.level().isClientSide && this.initialized && !this.getCharacterId().isEmpty()) {
   961→            dev.ergenverse.simulation.intent.ActorEntityLink.onEntitySpawn(this);
   962→        }
