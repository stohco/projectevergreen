   179→    private static RegistryObject<Block> registerAnchor(String name, BlockBehaviour.Properties props) {
   180→        RegistryObject<Block> block = BLOCKS.register(name, () -> new Block(props));
   181→        BLOCK_ITEMS.register(name, () -> new BlockItem(block.get(),
   182→                new Item.Properties()));
   183→        return block;
   184→    }
   185→
   186→    /**
   187→     * Register a workstation block + its BlockItem. Uses a custom Block
   188→     * supplier (e.g., {@link AlchemyFurnaceBlock::new}) so directional /
   189→     * block-entity-bearing blocks can be instantiated with the properties.
   190→     */
   191→    private static RegistryObject<Block> registerWorkstation(String name,
   192→                                                              BlockBehaviour.Properties props,
   193→                                                              java.util.function.Function<BlockBehaviour.Properties, Block> factory) {
   194→        RegistryObject<Block> block = BLOCKS.register(name, () -> factory.apply(props));
   195→        BLOCK_ITEMS.register(name, () -> new BlockItem(block.get(),
   196→                new Item.Properties()));
   197→        return block;
   198→    }
   199→
   200→    private static RegistryObject<Block> registerHerb(String name,
   201→                                                      @javax.annotation.Nullable net.minecraft.world.effect.MobEffect stewEffect,
   202→                                                      int effectDuration) {
   203→        // FIX CRON-COMPLETIONIST-65: Previously passed `null` as the Supplier when
   204→        // stewEffect was null, causing NPE during block registration (FlowerBlock
   205→        // constructor calls .get() on the Supplier). Now always passes a non-null
   206→        // Supplier that may return null (harmless — just no stew effect applied).
   207→        RegistryObject<Block> block = BLOCKS.register(name, () -> new FlowerBlock(
   208→                () -> stewEffect, effectDuration,
   209→                BlockBehaviour.Properties.of()
   210→                        .noCollission()
   211→                        .instabreak()
   212→                        .sound(SoundType.GRASS)
   213→                        .offsetType(BlockBehaviour.OffsetType.XZ)));
   214→        BLOCK_ITEMS.register(name, () -> new BlockItem(block.get(),
   215→                new Item.Properties()));
   216→        return block;
   217→    }

   275→        RegistryObject<Block> block = BLOCKS.register(name, () -> new Block(
   276→                BlockBehaviour.Properties.of()

   280→        return block;
