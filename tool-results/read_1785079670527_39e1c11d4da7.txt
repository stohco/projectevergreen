   208→     * Right-click interaction — dispatches to one of three behaviors based
   209→     * on context:
   210→     *
   211→     * <ol>
   212→     *   <li><b>If the off-hand holds an Element or Dao essence</b> →
   213→     *       absorb the essence into the bead (CRON-COMPLETIONIST-95).
   214→     *       This is the active side of bead progression — the player
   215→     *       aligns the 9 Parts by feeding essences to the bead. Consumes
   216→     *       one essence item per right-click.</li>
   217→     *   <li><b>Else if shift is held and the stage allows physical entry</b> →
   218→     *       teleport into the bead's interior dimension (existing behavior).</li>
   219→     *   <li><b>Else if the stage allows storage access</b> →
   220→     *       open the divine-sense storage menu (existing behavior).</li>
   221→     * </ol>
   222→     *
   223→     * <p>The essence-absorption branch takes priority over menu open so
   224→     * the player can absorb without accidental menu popups. To open the
   225→     * menu, the player must empty their off-hand first.
   226→     */
   227→    @Override
   228→    public InteractionResultHolder<ItemStack> use(net.minecraft.world.level.Level level,
   229→                                                   Player player, InteractionHand hand) {
   230→        ItemStack stack = player.getItemInHand(hand);
   231→
   232→        if (level.isClientSide()) {
   233→            return InteractionResultHolder.success(stack);
   234→        }
   235→
   236→        BeadInteriorStage stage = getStage(stack);
   237→
   238→        // DORMANT_STONE: nothing happens — it's just a stone.
   239→        if (stage == BeadInteriorStage.DORMANT_STONE) {
   240→            player.sendSystemMessage(
   241→                    Component.literal("The bead is cold and lifeless. A faint pattern "
   242→                            + "of five elements is barely visible on its surface.")
   243→                            .withStyle(ChatFormatting.GRAY));
   244→            return InteractionResultHolder.fail(stack);
   245→        }
   246→
   247→        // CRON-COMPLETIONIST-95: Essence absorption (off-hand priority).
   248→        // If the off-hand holds an Element or Dao essence, absorb it into
   249→        // the bead. This takes priority over menu open and dimension entry.
   250→        ItemStack offHand = player.getItemInHand(InteractionHand.OFF_HAND);
   251→        if (!offHand.isEmpty()) {
   252→            EssenceType essence = identifyEssence(offHand);
   253→            if (essence != null) {
   254→                return tryAbsorbEssence(level, player, hand, stack, offHand, essence);
   255→            }
   256→        }
   257→
   258→        if (player.isShiftKeyDown() && stage.hasPhysicalEntry) {
   259→            // Physical entry into the bead dimension
   260→            if (player instanceof ServerPlayer serverPlayer) {
   261→                enterBeadDimension(serverPlayer, stack);
   262→            }
   263→            return InteractionResultHolder.success(stack);
   264→        }
   265→
   266→        // Divine-sense access: open the storage menu
   267→        if (stage.hasStorageAccess) {
   268→            if (player instanceof ServerPlayer serverPlayer) {
   269→                openBeadMenu(serverPlayer, stack);
   270→            }
   271→            return InteractionResultHolder.success(stack);
   272→        }
   273→
   274→        // Should not happen (CRACK_OPENED and SMALL_SPACE both have storage),
   275→        // but handle gracefully.
   276→        player.sendSystemMessage(
   277→                Component.literal("The bead pulses faintly but you cannot yet "
   278→                        + "comprehend how to access it.")
   279→                        .withStyle(ChatFormatting.DARK_GRAY));
   280→        return InteractionResultHolder.fail(stack);
   281→    }

   334→    /**
   335→     * Attempt to absorb one essence from the off-hand into the bead.
   336→     *
   337→     * <p>Behavior:
   338→     * <ol>
   339→     *   <li>If the corresponding Part is already aligned → fail with a
   340→     *       chat message ("This essence has already been absorbed.").
   341→     *       The essence is NOT consumed.</li>
   342→     *   <li>Otherwise: align the Part (sets the bit + increments the
   343→     *       counter), consumes one essence from the off-hand, spawns
   344→     *       celebration particles, and sends a canon-flavored chat
   345→     *       message describing the absorption.</li>
   346→     * </ol>
   347→     */
   348→    private InteractionResultHolder<ItemStack> tryAbsorbEssence(
   349→            net.minecraft.world.level.Level level, Player player,
   350→            InteractionHand hand, ItemStack beadStack, ItemStack offHand,
   351→            EssenceType essence) {
   352→
   353→        int partBitIndex = essence.part.ordinal();
   354→
   355→        // 1. Check if the Part is already aligned.
   356→        if (isPartAligned(beadStack, partBitIndex)) {
   357→            player.sendSystemMessage(
   358→                    Component.literal("The bead rejects the " + essence.displayName
   359→                            + " — that pattern is already aligned.")
   360→                            .withStyle(ChatFormatting.YELLOW));
   361→            return InteractionResultHolder.fail(beadStack);
   362→        }
   363→
   364→        // 2. Canon ordering check: the Five Elements must be aligned
   365→        //    in order (Metal → Wood → Water → Fire → Earth). The 3 hidden
   366→        //    fragments can only be aligned after all 5 Elements are done.
   367→        //    This is a soft canon-faithful gate — the novel depicts Wang
   368→        //    Lin aligning the Five Elements roughly in this order, and the
   369→        //    3 hidden fragments are late-game.
   370→        HeavenDefyingBead.Part[] order = HeavenDefyingBead.Part.values();
   371→        for (int i = 0; i < partBitIndex; i++) {
   372→            if (!isPartAligned(beadStack, i)) {
   373→                HeavenDefyingBead.Part prerequisite = order[i];
   374→                player.sendSystemMessage(
   375→                        Component.literal("The bead cannot yet accept the "
   376→                                + essence.displayName + ". The "
   377→                                + prerequisite.name
   378→                                + " must be aligned first.")
   379→                                .withStyle(ChatFormatting.RED));
   380→                return InteractionResultHolder.fail(beadStack);
   381→            }
   382→        }
   383→
   384→        // 3. Consume one essence from the off-hand.
   385→        if (!player.getAbilities().instabuild) {
   386→            offHand.shrink(1);
   387→        }
   388→
   389→        // 4. Align the Part (sets the bit + increments the counter).
   390→        alignPart(beadStack, partBitIndex);
   391→
   392→        // 5. Server-side feedback: chat message + particles.
   393→        if (player instanceof ServerPlayer serverPlayer) {
   394→            int newCount = getPartsAligned(beadStack);
   395→            serverPlayer.sendSystemMessage(
   396→                    Component.literal(essence.flavorText)
   397→                            .withStyle(ChatFormatting.GOLD));
   398→            serverPlayer.sendSystemMessage(
   399→                    Component.literal("  Parts Aligned: " + newCount + " / 9")
   400→                            .withStyle(ChatFormatting.AQUA));
   401→
   402→            // Spawn celebration particles around the player.
   403→            spawnAbsorptionParticles(serverPlayer);
   404→
   405→            Ergenverse.LOGGER.info("[Ergenverse] Player {} absorbed {} into the bead "
   406→                            + "(parts aligned: {}/9, part={}).",
   407→                    serverPlayer.getName().getString(), essence.displayName,
   408→                    newCount, essence.part.name());
   409→        }
   410→
   411→        return InteractionResultHolder.success(beadStack);
   412→    }

   414→    /**
   415→     * Spawn gold and dust particles around the player to celebrate the
   416→     * absorption. Visible at any light level.
   417→     */
   418→    private void spawnAbsorptionParticles(ServerPlayer player) {
   419→        if (!(player.level() instanceof net.minecraft.server.level.ServerLevel serverLevel)) return;
   420→        double x = player.getX();
   421→        double y = player.getY() + 1.0;  // chest height
   422→        double z = player.getZ();
   423→        // 20 gold particles in a sphere
   424→        for (int i = 0; i < 20; i++) {
   425→            double theta = (i / 20.0) * Math.PI * 2;
   426→            double phi = ((i * 13) % 7) / 7.0 * Math.PI;  // pseudo-random sphere
   427→            double r = 1.2;
   428→            double dx = Math.sin(phi) * Math.cos(theta) * r;
   429→            double dy = Math.cos(phi) * r;
   430→            double dz = Math.sin(phi) * Math.sin(theta) * r;
   431→            serverLevel.sendParticles(ParticleTypes.END_ROD,
   432→                    x + dx, y + dy, z + dz, 1,
   433→                    0.0, 0.0, 0.0, 0.0);
   434→        }

   441→     * Returns true if the Part at {@code partIndex} (0..8) is aligned.
   442→     */
   443→    public boolean isPartAligned(ItemStack stack, int partIndex) {
   444→        if (partIndex < 0 || partIndex >= 9) return false;
   445→        if (stack.isEmpty() || !stack.hasTag()) return false;
   446→        int bits = stack.getTag().getInt(NBT_ALIGNED_PARTS);
   447→        return (bits & (1 << partIndex)) != 0;
   448→    }
   449→
   450→    /**
   451→     * Align the Part at {@code partIndex} (0..8). Sets the bit and
   452→     * increments the parts-aligned counter. Does NOT recompute the stage
   453→     * (the counter setter does that).
   454→     */
   455→    public void alignPart(ItemStack stack, int partIndex) {
   456→        if (partIndex < 0 || partIndex >= 9) return;
   457→        int bits = stack.getOrCreateTag().getInt(NBT_ALIGNED_PARTS);
   458→        if ((bits & (1 << partIndex)) != 0) return;  // already aligned
   459→        bits |= (1 << partIndex);
   460→        stack.getOrCreateTag().putInt(NBT_ALIGNED_PARTS, bits);
   461→        // Increment the parts-aligned counter (which calls recalculateStage).
   462→        setPartsAligned(stack, getPartsAligned(stack) + 1);
   463→    }

   470→        if (stack.isEmpty() || !stack.hasTag()) return 0;

   932→    private void recalculateStage(ItemStack stack) {
   933→        double stability = getSpatialStability(stack) / 10000.0;
   934→        double restoration = getPartsAligned(stack) / 9.0;
   935→        double authority = getOwnerAuthority(stack) / 10000.0;
   936→        double interior = getInteriorGrowth(stack) / 10000.0;
   937→        BeadInteriorStage newStage = BeadCapacityModel.stageFor(
   938→                stability, restoration, authority, interior);
   939→        setStage(stack, newStage);
   940→    }

   942→    /**
   943→     * Initialize a fresh bead stack to CRACK_OPENED stage with Situ Nan.
   944→     * Called when the player first obtains the bead (e.g., via command or
   945→     * the Heng Yue Sect discovery event).
   946→     *
   947→     * <p>Canon: the bead starts as a stone (DORMANT_STONE). After Situ Nan
   948→     * blasts it open, it reveals a small interior chamber (CRACK_OPENED).
   949→     * This method simulates that first opening.
   950→     *
   951→     * <p><b>CRON-COMPLETIONIST-95 FIX:</b> Previously this returned
   952→     * {@link ItemStack#EMPTY}, leaving no programmatic way to create an
   953→     * initialized bead. Now resolves the bead item via {@link WangLinItems}
   954→     * and applies the opening event. Returns {@link ItemStack#EMPTY} only
   955→     * if the bead item is not yet registered (e.g., called before
   956→     * Forge registry event fires).
   957→     */
   958→    public static ItemStack createInitialBead() {
   959→        RegistryObject<Item> beadRO = WangLinItems.get("wanglin/heaven_defying_bead");
   960→        if (beadRO == null || !beadRO.isPresent()) {
   961→            Ergenverse.LOGGER.warn("[Ergenverse] HeavenDefyingBeadItem.createInitialBead: "
   962→                    + "heaven_defying_bead not registered — returning EMPTY.");
   963→            return ItemStack.EMPTY;
   964→        }
   965→        ItemStack stack = new ItemStack(beadRO.get());
   966→        applyInitialOpening(stack);
   967→        return stack;
   968→    }

   970→    /**
   971→     * Apply the initial opening event to an existing bead stack.
   972→     * Transitions from DORMANT_STONE to CRACK_OPENED with Situ Nan present.
   973→     *
   974→     * <p>Canon: Ch. 8 — Wang Lin finds the bead as a stone. Situ Nan's
   975→     * remnant soul is inside. The bead cracks open, revealing a small chamber.
   976→     *
   977→     * <p><b>CRON-COMPLETIONIST-95:</b> Also sets the CORE bit in the
   978→     * AlignedParts bitfield, so the player cannot accidentally absorb a
   979→     * "core" essence (there is none — the core is the bead itself).
   980→     */
   981→    public static void applyInitialOpening(ItemStack stack) {
   982→        if (!(stack.getItem() instanceof HeavenDefyingBeadItem beadItem)) return;
   983→
   984→        // CRON-95: align the CORE part (bit 0) BEFORE setting the counter,
   985→        // so the bitfield and counter stay in sync.
   986→        int bits = stack.getOrCreateTag().getInt(NBT_ALIGNED_PARTS);
   987→        bits |= (1 << HeavenDefyingBead.Part.CORE.ordinal());
   988→        stack.getOrCreateTag().putInt(NBT_ALIGNED_PARTS, bits);
   989→
   990→        beadItem.setPartsAligned(stack, 1);  // CORE only
   991→        beadItem.setSpatialStability(stack, 1000);  // unstable but present
   992→        beadItem.setOwnerAuthority(stack, 500);  // faint recognition
   993→        beadItem.setInteriorGrowth(stack, 0);
   994→        beadItem.setSpirit(stack, HeavenDefyingBead.Spirit.SITU_NAN);
   995→        // recalculateStage is called inside setPartsAligned
   996→    }
