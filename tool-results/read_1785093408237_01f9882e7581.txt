    92→ * <p>The redemption fires when the player right-clicks the
    93→ * {@link dev.ergenverse.block.CultivationPlanetCrystalBlock} (in the

   100→ *       inherits the Suzaku Son status (Ch 443-450+) BEFORE Wang Ping's
   101→ *       redemption (Ch 680+). The inheritance is a strict prerequisite.</li>

   103→ *       CRON-118 canon correction: the prior CRON-117 implementation
   104→ *       checked the bead's {@code isLiMuwanRevived} flag as a proxy for
   105→ *       "Wang Lin has the prerequisites to channel Ling Tianhou's sword
   106→ *       qi". This was chronologically INVERTED — Li Muwan is revived at
   107→ *       the END of the novel (Wang Lin at 踏天境), which is FAR AFTER
   108→ *       Wang Ping's redemption (Wang Lin at 问鼎). CRON-118 removes the
   109→ *       Li Muwan revived proxy and replaces it with the canon-faithful

   112→ *       in their inventory (canon: exactly two strands) to trigger the
   113→ *       redemption. The 2 strands are consumed on successful redemption.</li>

   314→    /**
   315→     * Redeem Wang Ping — clear his {@code deadUntilRevived} flag and
   316→     * materialize him as a mortal boy at the Suzaku Tomb.
   317→     *
   318→     * <p>Called by {@link dev.ergenverse.block.CultivationPlanetCrystalBlock#use}
   319→     * AFTER all prerequisites are verified (Crystal inherited, bead with
   320→     * Li Muwan revived in hand, player realm ≥ ASCENDANT, Wang Ping's
   321→     * deadUntilRevived flag still true). This method performs the
   322→     * canon-faithful redemption: clears the flag, materializes Wang Ping,
   323→     * spawns effects, displays the message, and records in HistoryManager.
   324→     *
   325→     * <p>Defensive: any failure (WorldRuntime unavailable, ActorState
   326→     * missing, materialization failure) is logged at WARN level and the
   327→     * method returns {@code false}. The caller (CultivationPlanetCrystalBlock)
   328→     * does NOT undo the right-click — the player can retry by right-clicking
   329→     * again. The redemption is idempotent: if it succeeds once, the flag
   330→     * is cleared and subsequent right-clicks fall through to the "Crystal
   331→     * is silent" message.
   332→     *
   333→     * @param player     the server player (Wang Lin) who triggered the
   334→     *                   redemption by right-clicking the Crystal
   335→     * @param crystalPos the position of the Cultivation Planet Crystal
   336→     *                   (Wang Ping materializes near this position)
   337→     * @param currentTick the current game tick (for HistoryManager)
   338→     * @return {@code true} if Wang Ping was successfully redeemed and
   339→     *         materialized; {@code false} on any failure (logged at WARN)
   340→     */
   341→    public static boolean redeemAtSuzakuTomb(ServerPlayer player,
   342→                                               BlockPos crystalPos,
   343→                                               long currentTick) {
   344→        ServerLevel level = player.serverLevel();
   345→
   346→        // 1. Get the WorldRuntime singleton (defensive — mirrors LiMuwanRevivalEvent).
   347→        WorldRuntime runtime;
   348→        try {
   349→            runtime = WorldRuntime.get();
   350→        } catch (Throwable t) {
   351→            Ergenverse.LOGGER.error("[Ergenverse] CRON-117: WorldRuntime not available — "
   352→                    + "cannot redeem Wang Ping. Error: {}", t.getMessage(), t);
   353→            player.sendSystemMessage(Component.literal(
   354→                    "王平的残魂在天逆珠中叹息——世界运行时尚未就绪。")
   355→                    .withStyle(ChatFormatting.RED));
   356→            return false;
   357→        }
   358→
   359→        // 2. Get Wang Ping's ActorState from the NPCRuntime.
   360→        NPCRuntime.ActorState state = runtime.npcs().getActor(CanonUUID.WANG_PING);
   361→        if (state == null) {
   362→            Ergenverse.LOGGER.error("[Ergenverse] CRON-117: Wang Ping's ActorState not found "
   363→                    + "in NPCRuntime (canon UUID {}). Cannot redeem.", CanonUUID.WANG_PING);
   364→            player.sendSystemMessage(Component.literal(
   365→                    "王平的数据未在世界中注册。")
   366→                    .withStyle(ChatFormatting.RED));
   367→            return false;
   368→        }
   369→
   370→        // 3. Idempotency guard: if Wang Ping is already redeemed (deadUntilRevived=false),
   371→        //    this method should not have been called. But defensive: check and return.
   372→        if (!state.deadUntilRevived) {
   373→            Ergenverse.LOGGER.warn("[Ergenverse] CRON-117: Wang Ping is already redeemed "
   374→                    + "(deadUntilRevived=false). The right-click should have fallen through "
   375→                    + "to the 'Crystal is silent' message. Treating as a no-op.");
   376→            return false;
   377→        }
   378→
   379→        // 4. Dematerialize existing Wang Ping entity (if any — defensive).
   380→        //    This shouldn't happen (Wang Ping is deadUntilRevived=true, so he
   381→        //    shouldn't be materialized), but if he is (e.g., from a prior
   382→        //    redemption that wasn't fully cleaned up), dematerialize before
   383→        //    re-spawning.
   384→        if (runtime.npcs().isMaterialized(CanonUUID.WANG_PING)) {
   385→            runtime.npcs().dematerializeActor(CanonUUID.WANG_PING, runtime);
   386→            Ergenverse.LOGGER.info("[Ergenverse] CRON-117: Dematerialized existing Wang Ping "
   387→                    + "entity before re-spawning at the Suzaku Tomb.");
   388→        }
   389→
   390→        // 5. Update Wang Ping's ActorState to the Crystal's position.
   391→        //    Canon (mod-original condensation): the redemption happens at the
   392→        //    Suzaku Tomb (the conception site), NOT on Ranyun Star as in canon.
   393→        //    Future chunk reloads will re-materialize Wang Ping at this position.
   394→        state.x = crystalPos.getX();
   395→        state.z = crystalPos.getZ();
   396→
   397→        // 6. CRON-117: persist the redeemed state and clear the deadUntilRevived
   398→        //    flag. Mirrors the CRON-103 Li Muwan pattern + CRON-107 Tuo Sen pattern:
   399→        //    write the UUID to the WorldDeltaStore's revived-actor set (persists
   400→        //    across world reload), and clear the in-memory deadUntilRevived flag.
   401→        //    On world reload, WorldRuntime.initialize applies the revived set to
   402→        //    keep Wang Ping alive (cleared flag) across reloads.
   403→        try {
   404→            runtime.deltaStore().markActorRevived(CanonUUID.WANG_PING);
   405→            runtime.npcs().markActorAlive(CanonUUID.WANG_PING);
   406→            Ergenverse.LOGGER.info("[Ergenverse] CRON-117: marked Wang Ping as redeemed in "
   407→                    + "WorldDeltaStore (persisted) and cleared deadUntilRevived flag in NPCRuntime.");
   408→        } catch (Throwable t) {
   409→            Ergenverse.LOGGER.warn("[Ergenverse] CRON-117: failed to persist redeemed-actor "
   410→                    + "state for Wang Ping: {}", t.getMessage());
   411→            // Non-fatal: we still attempt the materialization below. If it
   412→            // fails because deadUntilRevived is still true, the failure path
   413→            // below handles it. The next right-click will retry the persist.
   414→        }
   415→

   522→    /**
   523→     * Find an entity by Minecraft entity ID in the server level.
   524→     * (Helper — ServerLevel has no direct getEntity(int) in 1.20.1; iteration
   525→     * is the supported approach. The entity count is small at spawn time.)
   526→     * Mirrors the {@link LiMuwanRevivalEvent#findEntityById} pattern.
   527→     *
   528→     * @param level    the server level
   529→     * @param entityId the Minecraft entity ID
   530→     * @return the EntityCultivator, or null if not found
   531→     */
   532→    private static EntityCultivator findEntityById(ServerLevel level, int entityId) {
   533→        for (net.minecraft.world.entity.Entity e : level.getAllEntities()) {
   534→            if (e.getId() == entityId && e instanceof EntityCultivator ec) {
   535→                return ec;
   536→            }
   537→        }
   538→        return null;
   539→    }

   541→    /**
   542→     * Spawn canon-faithful particle + sound effects for the redemption.
   543→     * The visual idiom (sword-qi coalescing into a body):
   544→     * <ul>
   545→     *   <li><b>END_ROD descending spiral at the Crystal</b> — sword qi
   546→     *       condensing downward from the Crystal into the world.</li>
   547→     *   <li><b>END_ROD ring at Wang Ping's spawn point</b> — the body
   548→     *       coalescing from the descending sword qi.</li>
   549→     *   <li><b>END_ROD stream from Crystal to Wang Ping</b> — the sword
   550→     *       qi flowing from the spiritual core to the body being built.</li>
   551→     *   <li><b>FIREWORK central flash at Wang Ping's chest</b> — the
   552→     *       moment of "False Life" ignition.</li>
   553→     *   <li><b>AMETHYST_BLOCK_CHIME at both ends</b> — the crystalline
   554→     *       tone of sword qi (Ling Tianhou is the Sword Venerable; his
   555→     *       sword qi manifests as a crystalline tone).</li>
   556→     *   <li><b>BELL at midpoint</b> — the formal moment of redemption
   557→     *       (like a temple bell, mirroring MuBingmeiAcceptanceEvent).</li>
   558→     *   <li><b>PLAYER_LEVELUP at Wang Ping</b> — the bright ascending
   559→     *       tone of new life (a child's first breath).</li>
   560→     * </ul>
   561→     *
   562→     * @param level     the server level
   563→     * @param crystalPos the Crystal's block position (sword-qi source)
   564→     * @param wangPing  the materialized Wang Ping entity (sword-qi destination)
   565→     */
   566→    private static void spawnRedemptionEffects(ServerLevel level,
   567→                                                 BlockPos crystalPos,
   568→                                                 EntityCultivator wangPing) {
   569→        double crystalX = crystalPos.getX() + 0.5;
   570→        double crystalY = crystalPos.getY() + 1.0;  // chest height
   571→        double crystalZ = crystalPos.getZ() + 0.5;
   572→
   573→        double bodyX = wangPing.getX();
   574→        double bodyY = wangPing.getY() + BODY_FORMATION_HEIGHT;
   575→        double bodyZ = wangPing.getZ();
   576→
   577→        double midX = (crystalX + bodyX) / 2.0;
   578→        double midY = (crystalY + bodyY) / 2.0;
   579→        double midZ = (crystalZ + bodyZ) / 2.0;
   580→
   581→        // ── Descending END_ROD spiral at the Crystal (sword qi condensing) ──
   582→        // A spiral that descends from 4 blocks above the Crystal to chest height,
   583→        // representing sword qi condensing downward into the world.
   584→        for (int i = 0; i < CRYSTAL_SPIRAL_PARTICLE_COUNT; i++) {
   585→            double t = i / (double) CRYSTAL_SPIRAL_PARTICLE_COUNT;
   586→            double theta = t * Math.PI * 4;  // 2 full turns
   587→            double radius = 0.8 * (1.0 - t * 0.5);  // narrowing spiral
   588→            double dx = Math.cos(theta) * radius;
   589→            double dz = Math.sin(theta) * radius;
   590→            // Descending from 4 blocks above to chest height
   591→            double dy = 4.0 * (1.0 - t);
   592→            level.sendParticles(ParticleTypes.END_ROD,
   593→                    crystalX + dx, crystalPos.getY() + dy, crystalZ + dz, 1,
   594→                    0.0, -0.08, 0.0, 0.0);
   595→        }
   596→
   597→        // ── Body-formation END_ROD ring at Wang Ping (body coalescing) ──
   598→        // A ring at chest height, expanding outward — the body forming.
   599→        for (int i = 0; i < BODY_RING_PARTICLE_COUNT; i++) {
   600→            double theta = (i / (double) BODY_RING_PARTICLE_COUNT) * Math.PI * 2;
   601→            double radius = 0.6;
   602→            double dx = Math.cos(theta) * radius;
   603→            double dz = Math.sin(theta) * radius;
   604→            // Slight vertical jitter for a "body forming" effect
   605→            double dy = (i % 3 - 1) * 0.15;
   606→            level.sendParticles(ParticleTypes.END_ROD,
   607→                    bodyX + dx, bodyY + dy, bodyZ + dz, 1,
   608→                    0.0, 0.02, 0.0, 0.0);
   609→        }
   610→
   611→        // ── END_ROD stream from Crystal to Wang Ping (sword qi flowing) ──
   612→        for (int i = 0; i < SWORD_QI_STREAM_PARTICLE_COUNT; i++) {
   613→            double t = i / (double) (SWORD_QI_STREAM_PARTICLE_COUNT - 1);
   614→            double px = crystalX + (bodyX - crystalX) * t;
   615→            double py = crystalY + (bodyY - crystalY) * t;
   616→            double pz = crystalZ + (bodyZ - crystalZ) * t;
   617→            // Small lateral jitter for visual richness
   618→            double jitterX = (i % 3 - 1) * 0.08;
   619→            double jitterZ = (i % 5 - 2) * 0.08;
   620→            level.sendParticles(ParticleTypes.END_ROD,
   621→                    px + jitterX, py, pz + jitterZ, 1,
   622→                    0.0, 0.0, 0.0, 0.0);
   623→        }
   624→
   625→        // ── AMETHYST crystalline burst at both ends (sword qi tone) ──
   626→        level.sendParticles(ParticleTypes.END_ROD,
   627→                crystalX, crystalY, crystalZ, CRYSTAL_BURST_PARTICLE_COUNT,
   628→                0.4, 0.4, 0.4, 0.05);
   629→        level.sendParticles(ParticleTypes.END_ROD,
   630→                bodyX, bodyY, bodyZ, CRYSTAL_BURST_PARTICLE_COUNT,
   631→                0.4, 0.4, 0.4, 0.05);
   632→
   633→        // ── Central FIREWORK flash at Wang Ping's chest (False Life igniting) ──
   634→        level.sendParticles(ParticleTypes.FIREWORK,
   635→                bodyX, bodyY, bodyZ, CENTRAL_FLASH_PARTICLE_COUNT,
   636→                0.3, 0.3, 0.3, 0.05);
   637→
   638→        // ── Sounds ──
   639→        // AMETHYST_BLOCK_CHIME at the Crystal (crystalline sword-qi source)
   640→        level.playSound(null, crystalPos,
   641→                SoundEvents.AMETHYST_BLOCK_CHIME, SoundSource.AMBIENT,
   642→                0.9F, 1.0F);  // volume 0.9, pitch 1.0 (clear crystalline tone)
   643→
   644→        // AMETHYST_BLOCK_CHIME at Wang Ping (crystalline sword-qi coalescing)
   645→        level.playSound(null, wangPing.blockPosition(),
   646→                SoundEvents.AMETHYST_BLOCK_CHIME, SoundSource.AMBIENT,
   647→                0.8F, 1.3F);  // volume 0.8, pitch 1.3 (slightly brighter at body)
   648→
   649→        // PLAYER_LEVELUP at Wang Ping (bright ascending — new life)
   650→        level.playSound(null, wangPing.blockPosition(),
   651→                SoundEvents.PLAYER_LEVELUP, SoundSource.AMBIENT,
   652→                0.7F, 1.4F);  // volume 0.7, pitch 1.4 (bright, ascending)
   653→
   654→        // BELL at midpoint (formal redemption — like a temple bell)
   655→        level.playSound(null, BlockPos.containing(midX, midY, midZ),
   656→                SoundEvents.BELL_BLOCK, SoundSource.AMBIENT,
   657→                1.0F, 1.0F);  // volume 1.0, pitch 1.0 (clear bell tone)
   658→    }
