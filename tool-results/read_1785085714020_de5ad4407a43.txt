   246→    public static void tick(ServerLevel level, long currentTick) {
   247→        // Gate 1: throttle to every ACCEPTANCE_CHECK_INTERVAL_TICKS
   248→        if (currentTick % ACCEPTANCE_CHECK_INTERVAL_TICKS != 0) return;
   249→        // Gate 2: server-side only
   250→        if (level == null || level.isClientSide()) return;
   251→
   252→        try {
   253→            tickImpl(level, currentTick);
   254→        } catch (Throwable t) {
   255→            // Defensive: never crash the server tick loop.
   256→            Ergenverse.LOGGER.error("[Ergenverse] CRON-113: MuBingmeiAcceptanceEvent "
   257→                    + "threw an exception: {}", t.getMessage(), t);
   258→        }
   259→    }

   261→    private static void tickImpl(ServerLevel level, long currentTick) {
   262→        // ── Gate 3: Find 周茹's materialized EntityCultivator ──
   263→        EntityCultivator zhouRu = findCultivatorByCharacterId(level, ZHOU_RU_CHARACTER_ID);
   264→        if (zhouRu == null) {
   265→            // 周茹 not materialized (chunk not loaded) — no-op.
   266→            return;
   267→        }
   268→
   269→        // ── Gate 4: CRON-112 prerequisite — sent_to_kunxu must be true ──
   270→        WorldRuntimeState runtime = WorldRuntimeState.get(level);
   271→        CompoundTag zhouRuState = runtime.getNpcState(ZHOU_RU_CHARACTER_ID);
   272→        if (zhouRuState == null) {
   273→            // No runtime state at all — CRON-110/112 haven't fired. No-op.
   274→            return;
   275→        }
   276→        if (!zhouRuState.getBoolean("sent_to_kunxu")) {
   277→            // CRON-112's departure hasn't fired. No-op.
   278→            return;
   279→        }
   280→
   281→        // ── Gate 5: Write-once guard — accepted_as_disciple must be false ──
   282→        if (zhouRuState.getBoolean("accepted_as_disciple")) {
   283→            // Already accepted — one-time event per save. No-op.
   284→            return;
   285→        }
   286→
   287→        // ── Gate 6: Find Mu Bingmei's materialized EntityCultivator ──
   288→        EntityCultivator muBingmei = findCultivatorByCharacterId(level, MU_BINGMEI_CHARACTER_ID);
   289→        if (muBingmei == null) {
   290→            // Mu Bingmei not materialized — no-op. The acceptance requires
   291→            // the master's presence.
   292→            return;
   293→        }
   294→
   295→        // ── Gate 7: Proximity check — 周茹 within 32 blocks of Mu Bingmei ──
   296→        double distSq = zhouRu.distanceToSqr(muBingmei);
   297→        if (distSq > ACCEPTANCE_PROXIMITY_RADIUS_SQ) {
   298→            // 周茹 too far from Mu Bingmei — no-op. The formal acceptance
   299→            // requires close proximity (a face-to-face ritual).
   300→            return;
   301→        }
   302→
   303→        // ── All gates pass — execute the acceptance ──
   304→
   305→        // ── Step 1: Set the runtime state flags ──

   307→        zhouRuState.putLong("disciple_acceptance_tick", currentTick);
   308→        zhouRuState.putString("master_character_id", MU_BINGMEI_CHARACTER_ID);
   309→        runtime.updateNpcState(ZHOU_RU_CHARACTER_ID, zhouRuState);
   310→
   311→        // ── Step 2: Spawn particle + sound effects ──
   312→        spawnAcceptanceEffects(level, zhouRu, muBingmei);
   313→
   314→        // ── Step 3: Find a nearby player to witness ──
   315→        ServerPlayer witness = findNearbyPlayer(level, zhouRu, WITNESS_RADIUS);
   316→
   317→        // ── Step 4: Display bilingual message (if witness present) ──
   318→        if (witness != null) {
   319→            announceAcceptance(witness, zhouRu, muBingmei);
   320→
   321→            // ── Step 5: Record in HistoryManager ──
   322→            HistoryManager.onDiscovery(witness,
   323→                    "mu_bingmei_accepts_zhou_ru_as_disciple",

   343→    /**
   344→     * Find a materialized EntityCultivator by characterId. Iterates the
   345→     * level's all-entities list (O(n) where n is the loaded entity count).
   346→     * Mirrors {@link ZhouRuCultivationGrowthService#findCultivatorByCharacterId}.
   347→     *
   348→     * @param level       the server level
   349→     * @param characterId the canon character id (e.g. "zhou_ru", "mu_bingmei")
   350→     * @return the EntityCultivator, or null if not found / not materialized
   351→     */
   352→    private static EntityCultivator findCultivatorByCharacterId(ServerLevel level,
   353→                                                                 String characterId) {
   354→        for (Entity e : level.getAllEntities()) {
   355→            if (!(e instanceof EntityCultivator cultivator)) continue;
   356→            if (!e.isAlive()) continue;
   357→            if (characterId.equals(cultivator.getCharacterId())) {
   358→                return cultivator;
   359→            }
   360→        }
   361→        return null;
   362→    }

   364→    /**
   365→     * Find the nearest ServerPlayer within {@code radius} blocks of the
   366→     * given entity. Used to find a witness for the acceptance event
   367→     * (for the bilingual message + HistoryManager recording).
   368→     *
   369→     * <p>In single-player maximalism, there is exactly one player. This
   370→     * method is defensive for future MP support.
   371→     *
   372→     * @param level  the server level
   373→     * @param entity the entity to search around
   374→     * @param radius the search radius (blocks, Euclidean)
   375→     * @return the nearest ServerPlayer, or null if none in radius
   376→     */
   377→    private static ServerPlayer findNearbyPlayer(ServerLevel level,
   378→                                                  EntityCultivator entity,
   379→                                                  double radius) {
   380→        List<ServerPlayer> players = level.players();
   381→        ServerPlayer nearest = null;
   382→        double nearestDistSq = radius * radius;
   383→        for (ServerPlayer p : players) {
   384→            double dSq = p.distanceToSqr(entity);
   385→            if (dSq <= nearestDistSq) {
   386→                nearestDistSq = dSq;
   387→                nearest = p;
   388→            }
   389→        }
   390→        return nearest;
   391→    }
