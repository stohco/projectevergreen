   164→ * <h2>Drop Behavior</h2>
   165→ *
   166→ * <p>The Crystal does NOT drop as an item when broken. Canon: the Crystal

   169→ * the player breaks the Crystal block, it drops nothing. The block is

   294→    /**
   295→     * Spawn ambient END_ROD particles at the block's center.
   296→     *
   297→     * <p>Spawn rate: 1/{@value #PARTICLE_TICK_INTERVAL} ticks per block ≈
   298→     * once per second. The particle is {@link ParticleTypes#END_ROD} —
   299→     * small, white, drifting upward with built-in slow motion — the
   300→     * closest vanilla particle to "spiritual Qi emission."
   301→     *
   302→     * <p>This is CLIENT-ONLY. {@code animateTick} is only called on the
   303→     * client when the block is in view.
   304→     *
   305→     * @param state  the block state
   306→     * @param level  the level (client level)
   307→     * @param pos    the block position
   308→     * @param random client-side random source
   309→     */
   310→    @Override
   311→    public void animateTick(BlockState state, Level level, BlockPos pos, RandomSource random) {
   312→        // 1/N chance per tick — subtle, not flashy
   313→        if (random.nextInt(PARTICLE_TICK_INTERVAL) != 0) return;
   314→
   315→        // Spawn at the block's center, with a small random offset.
   316→        double x = pos.getX() + 0.5 + (random.nextDouble() - 0.5) * 0.3;
   317→        double y = pos.getY() + 0.7 + (random.nextDouble() - 0.5) * 0.3;
   318→        double z = pos.getZ() + 0.5 + (random.nextDouble() - 0.5) * 0.3;
   319→
   320→        // END_ROD particles drift slowly upward with built-in motion.
   321→        // The velocity here is a small nudge; the particle's built-in
   322→        // motion does most of the work.
   323→        double dx = (random.nextDouble() - 0.5) * 0.02;
   324→        double dy = 0.04 + random.nextDouble() * 0.02;
   325→        double dz = (random.nextDouble() - 0.5) * 0.02;
   326→
   327→        level.addParticle(ParticleTypes.END_ROD, x, y, z, dx, dy, dz);
   328→    }

   388→        if (playerRealm.order < MIN_INHERITANCE_REALM.order) {
   389→            serverPlayer.sendSystemMessage(Component.literal(
   390→                    "你的修为不足，无法承受星晶之力。需达"
   391→                            + MIN_INHERITANCE_REALM.nameCn
   392→                            + "（" + MIN_INHERITANCE_REALM.name + "）方能传承。")
   393→                    .withStyle(ChatFormatting.YELLOW));
   394→            serverPlayer.sendSystemMessage(Component.literal(
   395→                    "Your cultivation is insufficient to withstand the Crystal's power. "
   396→                            + "You must reach " + MIN_INHERITANCE_REALM.name
   397→                            + " (" + MIN_INHERITANCE_REALM.nameCn + ") to inherit.")

   418→                "─────────────────────────────────────")
   419→                .withStyle(ChatFormatting.LIGHT_PURPLE));
   420→        serverPlayer.sendSystemMessage(Component.literal(
   421→                "修炼星晶迸发出无尽灵气，灌入你的天逆珠。")
   422→                .withStyle(ChatFormatting.LIGHT_PURPLE, ChatFormatting.ITALIC));
   423→        serverPlayer.sendSystemMessage(Component.literal(
   424→                "你成为了第十五代朱雀子。")
   425→                .withStyle(ChatFormatting.GOLD, ChatFormatting.BOLD));
   426→        serverPlayer.sendSystemMessage(Component.literal(
   427→                "「朱雀星，我以王林之名，承此传承。」")

   429→        serverPlayer.sendSystemMessage(Component.literal(
   430→                "The Cultivation Planet Crystal erupts with boundless spiritual Qi, "
   431→                        + "pouring into your Heaven-Defying Bead.")
   432→                .withStyle(ChatFormatting.LIGHT_PURPLE, ChatFormatting.ITALIC));
   433→        serverPlayer.sendSystemMessage(Component.literal(
   434→                "You have become the 15th-generation Suzaku Son.")
   435→                .withStyle(ChatFormatting.GOLD, ChatFormatting.BOLD));
   436→        serverPlayer.sendSystemMessage(Component.literal(
   437→                "\"Planet Suzaku, by the name of Wang Lin, I accept this inheritance.\"")
   438→                .withStyle(ChatFormatting.GOLD, ChatFormatting.ITALIC));

   445→        HistoryManager.onDiscovery(serverPlayer, SUBJECT_SUZAKU_SON_INHERITANCE,

   462→    /**
   463→     * Find the Heaven-Defying Bead in the player's hands or inventory.
   464→     * Mirrors the {@link dev.ergenverse.wanglin.bead.LiMuwanSoulCaptureEvent#findBead}
   465→     * pattern (CRON-99): main hand → off hand → main inventory.
   466→     *
   467→     * @param player the player whose inventory to scan
   468→     * @return the bead ItemStack, or {@link ItemStack#EMPTY} if not found
   469→     */
   470→    private static ItemStack findBead(ServerPlayer player) {
   471→        // 1. Main hand
   472→        ItemStack mainHand = player.getMainHandItem();
   473→        if (mainHand.getItem() instanceof HeavenDefyingBeadItem) return mainHand;
   474→
   475→        // 2. Off hand
   476→        ItemStack offHand = player.getOffhandItem();
   477→        if (offHand.getItem() instanceof HeavenDefyingBeadItem) return offHand;
   478→
   479→        // 3. Main inventory (slots 0-35)
   480→        for (ItemStack stack : player.getInventory().items) {
   481→            if (stack.getItem() instanceof HeavenDefyingBeadItem) return stack;
   482→        }
   483→
   484→        return ItemStack.EMPTY;
   485→    }

   489→     * Mirrors the {@link dev.ergenverse.wanglin.ai.reasoning.ExpectationModelObserver#getPlayerRealm}
   490→     * pattern (CRON-99). Returns {@link RealmId#MORTAL} if the capability

   504→    /**
   505→     * Record a PLAYER delta for the block-state change, so the
   506→     * {@code inherited=true} state persists across chunk reloads.
   507→     *
   508→     * <p>The block-state change via {@code level.setBlock} is already
   509→     * applied to the world; this method ensures the change is ALSO
   510→     * recorded in the {@link WorldDeltaStore} journal, so the
   511→     * chunk-materializer re-applies it on reload.
   512→     *
   513→     * <p>Uses the CRON-94 {@link dev.ergenverse.runtime.delta.BlockStateCodec}
   514→     * format: {@code "ergenverse:cultivation_planet_crystal[inherited=true]"}.
   515→     * The codec preserves the block-state property across serialization,
   516→     * so the inherited flag survives chunk reload.
   517→     *
   518→     * <p>Defensive: silently no-ops if the WorldRuntime is not yet
   519→     * initialized (should not happen during gameplay, but defensive
   520→     * coding in case of registry ordering issues).
   521→     *
   522→     * @param pos   the block position
   523→     * @param state the new block state (with {@code inherited=true})
   524→     */
   525→    private static void recordPlayerDelta(BlockPos pos, BlockState state) {
   526→        try {
   527→            WorldRuntime runtime = WorldRuntime.get();
   528→            if (!runtime.isInitialized()) {
   529→                Ergenverse.LOGGER.warn("[Ergenverse] CRON-106: WorldRuntime not initialized; "
   530→                        + "Crystal inheritance at {} will not persist via journal.", pos);
   531→                return;
   532→            }
   533→            // Write the block-state change as a PLAYER delta via the WorldFacade.
   534→            // The facade records the delta in the WorldDeltaStore journal AND
   535→            // applies it to the world (the level.setBlock above already applied
   536→            // it; the facade call here is for journaling only — the duplicate
   537→            // world-write is a harmless no-op).
   538→            //
   539→            // The blockId string is the CRON-94 codec format:
   540→            //   "ergenverse:cultivation_planet_crystal[inherited=true]"
   541→            // which preserves the inherited property across serialization.
   542→            String blockId = dev.ergenverse.runtime.delta.BlockStateCodec.serialize(state);
   543→            runtime.world().setPlayerBlock(pos.getX(), pos.getY(), pos.getZ(), blockId);
   544→        } catch (Throwable t) {
   545→            Ergenverse.LOGGER.warn("[Ergenverse] CRON-106: Failed to record PLAYER delta "
   546→                    + "for Crystal inheritance at {}: {}", pos, t.getMessage());
   547→        }
   548→    }
