   251→     * {@link dev.ergenverse.core.Ergenverse#onServerTick} every server tick

   253→     * {@link #ACCEPTANCE_CHECK_INTERVAL_TICKS} — callers do NOT need to

   259→    public static void tick(ServerLevel level, long currentTick) {
   260→        // Gate 1: throttle to every ACCEPTANCE_CHECK_INTERVAL_TICKS
   261→        if (currentTick % ACCEPTANCE_CHECK_INTERVAL_TICKS != 0) return;
   262→        // Gate 2: server-side only
   263→        if (level == null || level.isClientSide()) return;
   264→
   265→        try {
   266→            tickImpl(level, currentTick);
   267→        } catch (Throwable t) {
   268→            // Defensive: never crash the server tick loop.
   269→            Ergenverse.LOGGER.error("[Ergenverse] CRON-113: MuBingmeiAcceptanceEvent "
   270→                    + "threw an exception: {}", t.getMessage(), t);
   271→        }
   272→    }

   276→        EntityCultivator zhouRu = findCultivatorByCharacterId(level, ZHOU_RU_CHARACTER_ID);

   285→        if (zhouRuState == null) {
   286→            // No runtime state at all — CRON-110/112 haven't fired. No-op.
   287→            return;
   288→        }
   289→        if (!zhouRuState.getBoolean("sent_to_kunxu")) {
   290→            // CRON-112's departure hasn't fired. No-op.
   291→            return;
   292→        }
   293→
   294→        // ── Gate 5: Write-once guard — accepted_as_disciple must be false ──
   295→        if (zhouRuState.getBoolean("accepted_as_disciple")) {
   296→            // Already accepted — one-time event per save. No-op.
   297→            return;
   298→        }

   300→        // ── Gate 6: Find Mu Bingmei's materialized EntityCultivator ──
   301→        EntityCultivator muBingmei = findCultivatorByCharacterId(level, MU_BINGMEI_CHARACTER_ID);
   302→        if (muBingmei == null) {
   303→            // Mu Bingmei not materialized — no-op. The acceptance requires

   306→        }

   308→        // ── Gate 7: Proximity check — 周茹 within 32 blocks of Mu Bingmei ──

   323→

   326→

   329→

   333→

   356→    /**
   357→     * Find a materialized EntityCultivator by characterId. Iterates the
   358→     * level's all-entities list (O(n) where n is the loaded entity count).
   359→     * Mirrors {@link ZhouRuCultivationGrowthService#findCultivatorByCharacterId}.
   360→     *
   361→     * @param level       the server level
   362→     * @param characterId the canon character id (e.g. "zhou_ru", "mu_bingmei")
   363→     * @return the EntityCultivator, or null if not found / not materialized
   364→     */
   365→    private static EntityCultivator findCultivatorByCharacterId(ServerLevel level,
   366→                                                                 String characterId) {
   367→        for (Entity e : level.getAllEntities()) {
   368→            if (!(e instanceof EntityCultivator cultivator)) continue;
   369→            if (!e.isAlive()) continue;
   370→            if (characterId.equals(cultivator.getCharacterId())) {
   371→                return cultivator;
   372→            }
   373→        }
   374→        return null;
   375→    }

   377→    /**
   378→     * Find the nearest ServerPlayer within {@code radius} blocks of the
   379→     * given entity. Used to find a witness for the acceptance event
   380→     * (for the bilingual message + HistoryManager recording).
   381→     *
   382→     * <p>In single-player maximalism, there is exactly one player. This
   383→     * method is defensive for future MP support.
   384→     *
   385→     * @param level  the server level
   386→     * @param entity the entity to search around
   387→     * @param radius the search radius (blocks, Euclidean)
   388→     * @return the nearest ServerPlayer, or null if none in radius
   389→     */
   390→    private static ServerPlayer findNearbyPlayer(ServerLevel level,
   391→                                                  EntityCultivator entity,
   392→                                                  double radius) {
   393→        List<ServerPlayer> players = level.players();
   394→        ServerPlayer nearest = null;
   395→        double nearestDistSq = radius * radius;
   396→        for (ServerPlayer p : players) {
   397→            double dSq = p.distanceToSqr(entity);
   398→            if (dSq <= nearestDistSq) {
   399→                nearestDistSq = dSq;
   400→                nearest = p;
   401→            }
   402→        }
   403→        return nearest;
   404→    }

   430→     *   <li>WITHER_SPAWN — too dark; Mu Bingmei's relationship with Wang Lin
   431→     *       is complex (she refined their son Wang Ping into a resentful
   432→     *       spirit), but the acceptance itself is a positive narrative beat.</li>
   433→     *   <li>DRAGON_BREATH particles — too aggressive; this is a sacred bond,
   434→     *       not a combat event.</li>
   435→     * </ul>
   436→     *
   437→     * @param level     the server level
   438→     * @param zhouRu    the disciple EntityCultivator (周茹)
   439→     * @param muBingmei the master EntityCultivator (木冰眉)
   440→     */
   441→    private static void spawnAcceptanceEffects(ServerLevel level,
   442→                                                EntityCultivator zhouRu,
   443→                                                EntityCultivator muBingmei) {
   444→        double masterX = muBingmei.getX();
   445→        double masterY = muBingmei.getY() + 1.0;  // chest height
   446→        double masterZ = muBingmei.getZ();
   447→
   448→        double discipleX = zhouRu.getX();
   449→        double discipleY = zhouRu.getY() + 1.0;  // chest height
   450→        double discipleZ = zhouRu.getZ();
   451→
   452→        double midX = (masterX + discipleX) / 2.0;
   453→        double midY = (masterY + discipleY) / 2.0;
   454→        double midZ = (masterZ + discipleZ) / 2.0;
   455→
   456→        // ── Master's Qi-descending ring (at Mu Bingmei) ──
   457→        for (int i = 0; i < MASTER_RING_PARTICLE_COUNT; i++) {
   458→            double theta = (i / (double) MASTER_RING_PARTICLE_COUNT) * Math.PI * 2;
   459→            double radius = 1.2;
   460→            double dx = Math.cos(theta) * radius;
   461→            double dz = Math.sin(theta) * radius;
   462→            // Descending from 3 blocks above head to chest height
   463→            double dy = 2.0 - ((i % 4) * 0.4);
   464→            level.sendParticles(ParticleTypes.END_ROD,
   465→                    masterX + dx, muBingmei.getY() + dy, masterZ + dz, 1,
   466→                    0.0, -0.05, 0.0, 0.0);
   467→        }
   468→
   469→        // ── Disciple's Qi-receiving ring (at 周茹) ──
   470→        for (int i = 0; i < DISCIPLE_RING_PARTICLE_COUNT; i++) {
   471→            double theta = (i / (double) DISCIPLE_RING_PARTICLE_COUNT) * Math.PI * 2;
   472→            double radius = 0.9;
   473→            double dx = Math.cos(theta) * radius;
   474→            double dz = Math.sin(theta) * radius;
   475→            // Ascending from feet to chest (receiving Qi upward)
   476→            double dy = 0.5 + ((i % 3) * 0.5);
   477→            level.sendParticles(ParticleTypes.END_ROD,
   478→                    discipleX + dx, zhouRu.getY() + dy, discipleZ + dz, 1,
   479→                    0.0, 0.05, 0.0, 0.0);
   480→        }
   481→
   482→        // ── Bond stream (Mu Bingmei → 周茹) ──
   483→        // A series of END_ROD particles along the line from master to disciple,
   484→        // simulating the Qi flowing from master to disciple.
   485→        for (int i = 0; i < BOND_STREAM_PARTICLE_COUNT; i++) {
   486→            double t = i / (double) (BOND_STREAM_PARTICLE_COUNT - 1);
   487→            double px = masterX + (discipleX - masterX) * t;
   488→            double py = masterY + (discipleY - masterY) * t;
   489→            double pz = masterZ + (discipleZ - masterZ) * t;
   490→            // Small lateral jitter for visual richness
   491→            double jitterX = (i % 3 - 1) * 0.1;
   492→            double jitterZ = (i % 5 - 2) * 0.1;
   493→            level.sendParticles(ParticleTypes.END_ROD,
   494→                    px + jitterX, py, pz + jitterZ, 1,
   495→                    0.0, 0.0, 0.0, 0.0);
   496→        }
   497→
   498→        // ── AMETHYST crystalline bond burst (at both ends) ──
   499→        level.sendParticles(ParticleTypes.END_ROD,
   500→                masterX, masterY, masterZ, CRYSTAL_BURST_PARTICLE_COUNT,
   501→                0.5, 0.5, 0.5, 0.05);
   502→        level.sendParticles(ParticleTypes.END_ROD,
   503→                discipleX, discipleY, discipleZ, CRYSTAL_BURST_PARTICLE_COUNT,
   504→                0.5, 0.5, 0.5, 0.05);
   505→
   506→        // ── Central FIREWORK flash at midpoint ──
   507→        level.sendParticles(ParticleTypes.FIREWORK,
   508→                midX, midY, midZ, CENTRAL_FLASH_PARTICLE_COUNT,
   509→                0.4, 0.4, 0.4, 0.05);
   510→
   511→        // ── Sounds ──
   512→        // AMETHYST_BLOCK_CHIME at both ends (crystalline bond)
   513→        level.playSound(null, muBingmei.blockPosition(),
   514→                SoundEvents.AMETHYST_BLOCK_CHIME, SoundSource.AMBIENT,
   515→                0.8F, 1.2F);  // volume 0.8, pitch 1.2 (crystalline)
   516→        level.playSound(null, zhouRu.blockPosition(),
   517→                SoundEvents.AMETHYST_BLOCK_CHIME, SoundSource.AMBIENT,
   518→                0.8F, 1.4F);  // volume 0.8, pitch 1.4 (slightly brighter at disciple)
   519→
   520→        // PLAYER_LEVELUP at 周茹 (bright ascending — the disciple's recognition)
   521→        level.playSound(null, zhouRu.blockPosition(),
   522→                SoundEvents.PLAYER_LEVELUP, SoundSource.AMBIENT,
   523→                0.6F, 1.3F);  // volume 0.6, pitch 1.3
   524→
   525→        // BELL at midpoint (formal acceptance — like a temple bell)
   526→        level.playSound(null, BlockPos.containing(midX, midY, midZ),
   527→                SoundEvents.BELL_BLOCK, SoundSource.AMBIENT,
   528→                0.9F, 1.0F);  // volume 0.9, pitch 1.0 (clear bell tone)
   529→    }
