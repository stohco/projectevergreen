     4→/*
     5→ * QilinModel — winged wolf-quadruped with antler horns and flowing mane.
     6→ *
     7→ * Canon (Renegade Immortal): qilin (麒麟) are divine beasts of extreme rarity,
     8→ * combining wolf/lion body with deer antlers, dragon scales, and spiritual
     9→ * authority. They appear at moments of great destiny.
    10→ *
    11→ * REAL QILIN MYTHOLOGY anatomy:
    12→ *   - Body: wolf/lion hybrid — deep chest, muscular haunches
    13→ *   - Head: wolf-like with dragon features, broad skull, noble bearing
    14→ *   - Horns: deer-like antlers (branched, multi-tine) — NOT simple horns
    15→ *   - Mane: flowing, sometimes flame-touched, along the neck and spine
    16→ *   - Tail: ox-like, long and tufted
    17→ *   - Legs: wolf paws, sometimes with scales
    18→ *   - Wings: some depictions show feathered wings (this model includes them)
    19→ *   - Scales: dragon scales shimmer on flanks
    20→ *
    21→ * This model extends SpiritWolfModel anatomy with:
    22→ *   - Antlers (branched 3-segment chain per side, not simple cones)
    23→ *   - Flowing mane (5 segments along neck/spine, like fire beast but longer)
    24→ *   - Tufted tail (3-segment chain with fan tip)
    25→ *   - GRAND FEATHERED WINGS (3-segment chain per side: shoulder→elbow→ primaries)
    26→ *     CRON-COMPLETIONIST-58: Upgraded from flat 5x0.6 box to 3-segment feathered
    27→ *     chain. Each wing has a shoulder bone (humerus), elbow joint, secondary
    28→ *     feathers, and 3 individual primary feathers that spread on flap.
    29→ *   - Scaled flank plates (2 boxes on each side of body)
    30→ *
    31→ * ANIMATION:
    32→ *   - Walk/Run: diagonal trot with spine flex (wolf base)
    33→ *   - Flight: wings extend and flap when POSE_FLYING
    34→ *   - Idle: mane sways, antlers still, tail slow sway
    35→ *   - Combat: head lowers, antlers thrust forward
    36→ *   - Death: collapse with quadratic ease-in
    37→ *
    38→ * HARSH SELF-CRITIQUE:
    39→ *   - FIXED (CRON-58): Wings upgraded from 1 flat 5x0.6 box to 3-segment feathered
    40→ *     chain. Still not real feathers, but now has distinct shoulder/secondary/primary
    41→ *     segments with independent flap animation. Scores 4/10 → 7/10 for wings.
    42→ *   - Antlers are boxy chains — real antlers are branching, tapering, irregular.
    43→ *     Each tine should be a different length and angle.
    44→ *   - Mane is box slabs — real flowing mane would need a chain of thin plates.
    45→ *   - Scales are flat boxes — no shimmer, no gradient, no overlap pattern.
    46→ *   - Tail fan is flat boxes — no hair detail, no tuft volume.
    47→ *   - PRIMARY FEATHERS are individual 1x0.2x3 boxes — they look like sticks.
    48→ *     Real primaries have width, curvature, and barb detail. The best we can do
    49→ *     with vanilla addBox API is thin rectangular slabs.
    50→ */

   369→    @Override

   371→                          float ageInTicks, float netHeadYaw, float headPitch) {
   372→        float yaw = netHeadYaw * 0.017453292F;
   373→        float pitch = headPitch * 0.017453292F;
   374→        this.head.yRot = Math.max(-1.0F, Math.min(1.0F, yaw));
   375→        this.head.xRot = Math.max(-0.7F, Math.min(0.7F, pitch));
   376→
   377→        boolean resting = entity.getSpiritPose() == SpiritBeastEntity.POSE_RESTING;
   378→        boolean flying = entity.getSpiritPose() == SpiritBeastEntity.POSE_FLYING;
   379→        boolean sprinting = entity.getSpiritPose() == SpiritBeastEntity.POSE_SPRINTING;
   380→        boolean swimming = entity.getSpiritPose() == SpiritBeastEntity.POSE_SWIMMING;
   381→
   382→        ModelPart[] mane = {this.mane0, this.mane1, this.mane2, this.mane3, this.mane4};
   383→
   384→        if (resting) {
   385→            float breath = (float) Math.sin(ageInTicks * 0.08F) * 0.08F;
   386→            this.root.y = -2.0F + breath;
   387→            this.frontLeftThigh.xRot = -0.6F;
   388→            this.frontRightThigh.xRot = -0.6F;
   389→            this.frontLeftShin.xRot = 0.4F;
   390→            this.frontRightShin.xRot = 0.4F;
   391→            this.backLeftThigh.xRot = 0.3F;
   392→            this.backRightThigh.xRot = 0.3F;
   393→            this.backLeftShin.xRot = -0.2F;
   394→            this.backRightShin.xRot = -0.2F;
   395→            this.head.xRot = 0.3F;
   396→            this.jaw.xRot = 0.0F;
   397→            // Mane settles
   398→            for (int i = 0; i < mane.length; i++) {
   399→                mane[i].yScale = 0.8F;
   400→                mane[i].yRot = (float) Math.sin(ageInTicks * 0.1F + i * 0.3F) * 0.05F;
   401→            }
   402→            this.tailBase.xRot = 0.5F;
   403→            // Wings tucked tight (rest pose)
   404→            this.leftWingRoot.zRot = 0.0F;
   405→            this.rightWingRoot.zRot = 0.0F;
   406→            this.leftWingRoot.xRot = -0.8F;
   407→            this.rightWingRoot.xRot = 0.8F;
   408→            this.leftWingMid.zRot = 0.0F;
   409→            this.rightWingMid.zRot = 0.0F;
   410→            this.leftWingTip.zRot = 0.0F;
   411→            this.rightWingTip.zRot = 0.0F;
   412→        } else if (swimming) {
   413→            // ── CRON-COMPLETIONIST-76: Qilin swims — divine beasts can ford rivers ──
   414→            // Canon: qilin are supernatural creatures that can walk on water or
   415→            // swim with serene grace. Body pitches, legs paddle, head elevated.
   416→            float paddle = ageInTicks * 1.0F;
   417→            float bob = (float) Math.sin(paddle * 0.5F) * 0.08F;
   418→            this.root.xRot = -0.25F;          // body pitches slightly down
   419→            this.root.y = -1.0F + bob;         // submerged partially
   420→            // Head stays above water
   421→            this.head.xRot = -0.3F;
   422→            this.neck.xRot = -0.5F;
   423→            this.jaw.xRot = 0.0F;
   424→            // Legs paddle — slower than wolf, more deliberate (divine poise)
   425→            this.frontLeftThigh.xRot  = (float) Math.cos(paddle) * 0.6F;
   426→            this.frontRightThigh.xRot = (float) Math.cos(paddle + Math.PI) * 0.6F;
   427→            this.backLeftThigh.xRot   = (float) Math.cos(paddle + Math.PI) * 0.5F;
   428→            this.backRightThigh.xRot  = (float) Math.cos(paddle) * 0.5F;
   429→            this.frontLeftShin.xRot  = -0.2F + Math.abs((float) Math.cos(paddle)) * 0.4F;
   430→            this.frontRightShin.xRot = -0.2F + Math.abs((float) Math.cos(paddle + Math.PI)) * 0.4F;
   431→            this.backLeftShin.xRot   = -0.2F + Math.abs((float) Math.cos(paddle + Math.PI)) * 0.3F;
   432→            this.backRightShin.xRot  = -0.2F + Math.abs((float) Math.cos(paddle)) * 0.3F;
   433→            // Tail sways in water — wider than on land
   434→            this.tailBase.yRot = (float) Math.sin(paddle * 0.5F) * 0.4F;
   435→            this.tailMid.yRot = (float) Math.sin(paddle * 0.5F + 0.5F) * 0.25F;
   436→            this.tailFan.yRot = (float) Math.sin(paddle * 0.5F + 1.0F) * 0.2F;
