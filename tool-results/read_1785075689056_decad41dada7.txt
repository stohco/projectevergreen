   179→   179→    private static RegistryObject<Block> registerAnchor(String name, BlockBehaviour.Properties props) {
   180→   180→        RegistryObject<Block> block = BLOCKS.register(name, () -> new Block(props));
   181→   181→        BLOCK_ITEMS.register(name, () -> new BlockItem(block.get(),
   182→   182→                new Item.Properties()));
   183→   183→        return block;
   184→   184→    }
   185→   185→
   186→   186→    /**
   187→   187→     * Register a workstation block + its BlockItem. Uses a custom Block
   188→   188→     * supplier (e.g., {@link AlchemyFurnaceBlock::new}) so directional /
   189→   189→     * block-entity-bearing blocks can be instantiated with the properties.
   190→   190→     */
   191→   191→    private static RegistryObject<Block> registerWorkstation(String name,
   192→   192→                                                              BlockBehaviour.Properties props,
   193→   193→                                                              java.util.function.Function<BlockBehaviour.Properties, Block> factory) {
   194→   194→        RegistryObject<Block> block = BLOCKS.register(name, () -> factory.apply(props));
   195→   195→        BLOCK_ITEMS.register(name, () -> new BlockItem(block.get(),
   196→   196→                new Item.Properties()));
   197→   197→        return block;
   198→   198→    }
   199→   199→
   200→   200→    private static RegistryObject<Block> registerHerb(String name,
   201→   201→                                                      @javax.annotation.Nullable net.minecraft.world.effect.MobEffect stewEffect,
   202→   202→                                                      int effectDuration) {
   203→   203→        // FIX CRON-COMPLETIONIST-65: Previously passed `null` as the Supplier when
   204→   204→        // stewEffect was null, causing NPE during block registration (FlowerBlock
   205→   205→        // constructor calls .get() on the Supplier). Now always passes a non-null
   206→   206→        // Supplier that may return null (harmless — just no stew effect applied).
   207→   207→        RegistryObject<Block> block = BLOCKS.register(name, () -> new FlowerBlock(
   208→   208→                () -> stewEffect, effectDuration,
   209→   209→                BlockBehaviour.Properties.of()
   210→   210→                        .noCollission()
   211→   211→                        .instabreak()
   212→   212→                        .sound(SoundType.GRASS)
   213→   213→                        .offsetType(BlockBehaviour.OffsetType.XZ)));
   214→   214→        BLOCK_ITEMS.register(name, () -> new BlockItem(block.get(),
   215→   215→                new Item.Properties()));
   216→   216→        return block;
   217→   217→    }
