   169→ * <p>The Crystal does NOT drop as an item when broken. Canon: the Crystal

   313→    @Override
   314→    public void animateTick(BlockState state, Level level, BlockPos pos, RandomSource random) {
   315→        // 1/N chance per tick — subtle, not flashy
   316→        if (random.nextInt(PARTICLE_TICK_INTERVAL) != 0) return;
   317→
   318→        // Spawn at the block's center, with a small random offset.
   319→        double x = pos.getX() + 0.5 + (random.nextDouble() - 0.5) * 0.3;
   320→        double y = pos.getY() + 0.7 + (random.nextDouble() - 0.5) * 0.3;
   321→        double z = pos.getZ() + 0.5 + (random.nextDouble() - 0.5) * 0.3;
   322→
   323→        // END_ROD particles drift slowly upward with built-in motion.
   324→        // The velocity here is a small nudge; the particle's built-in
   325→        // motion does most of the work.
   326→        double dx = (random.nextDouble() - 0.5) * 0.02;
   327→        double dy = 0.04 + random.nextDouble() * 0.02;
   328→        double dz = (random.nextDouble() - 0.5) * 0.02;
   329→
   330→        level.addParticle(ParticleTypes.END_ROD, x, y, z, dx, dy, dz);
   331→    }

   484→    /**
   485→     * Find the Heaven-Defying Bead in the player's hands or inventory.
   486→     * Mirrors the {@link dev.ergenverse.wanglin.bead.LiMuwanSoulCaptureEvent#findBead}
   487→     * pattern (CRON-99): main hand → off hand → main inventory.
   488→     *
   489→     * @param player the player whose inventory to scan
   490→     * @return the bead ItemStack, or {@link ItemStack#EMPTY} if not found
   491→     */
   492→    private static ItemStack findBead(ServerPlayer player) {
   493→        // 1. Main hand
   494→        ItemStack mainHand = player.getMainHandItem();
   495→        if (mainHand.getItem() instanceof HeavenDefyingBeadItem) return mainHand;
   496→
   497→        // 2. Off hand
   498→        ItemStack offHand = player.getOffhandItem();
   499→        if (offHand.getItem() instanceof HeavenDefyingBeadItem) return offHand;
   500→
   501→        // 3. Main inventory (slots 0-35)
   502→        for (ItemStack stack : player.getInventory().items) {
   503→            if (stack.getItem() instanceof HeavenDefyingBeadItem) return stack;
   504→        }
   505→
   506→        return ItemStack.EMPTY;
   507→    }

   526→    /**
   527→     * Record a PLAYER delta for the block-state change, so the
   528→     * {@code inherited=true} state persists across chunk reloads.
   529→     *
   530→     * <p>The block-state change via {@code level.setBlock} is already
   531→     * applied to the world; this method ensures the change is ALSO
   532→     * recorded in the {@link WorldDeltaStore} journal, so the
   533→     * chunk-materializer re-applies it on reload.
   534→     *
   535→     * <p>Uses the CRON-94 {@link dev.ergenverse.runtime.delta.BlockStateCodec}
   536→     * format: {@code "ergenverse:cultivation_planet_crystal[inherited=true]"}.
   537→     * The codec preserves the block-state property across serialization,
   538→     * so the inherited flag survives chunk reload.
   539→     *
   540→     * <p>Defensive: silently no-ops if the WorldRuntime is not yet
   541→     * initialized (should not happen during gameplay, but defensive
   542→     * coding in case of registry ordering issues).
   543→     *
   544→     * @param pos   the block position
   545→     * @param state the new block state (with {@code inherited=true})
   546→     */
   547→    private static void recordPlayerDelta(BlockPos pos, BlockState state) {
   548→        try {
   549→            WorldRuntime runtime = WorldRuntime.get();
   550→            if (!runtime.isInitialized()) {
   551→                Ergenverse.LOGGER.warn("[Ergenverse] CRON-106: WorldRuntime not initialized; "
   552→                        + "Crystal inheritance at {} will not persist via journal.", pos);
   553→                return;
   554→            }
   555→            // Write the block-state change as a PLAYER delta via the WorldFacade.
   556→            // The facade records the delta in the WorldDeltaStore journal AND
   557→            // applies it to the world (the level.setBlock above already applied
   558→            // it; the facade call here is for journaling only — the duplicate
   559→            // world-write is a harmless no-op).
   560→            //
   561→            // The blockId string is the CRON-94 codec format:
   562→            //   "ergenverse:cultivation_planet_crystal[inherited=true]"
   563→            // which preserves the inherited property across serialization.
   564→            String blockId = dev.ergenverse.runtime.delta.BlockStateCodec.serialize(state);
   565→            runtime.world().setPlayerBlock(pos.getX(), pos.getY(), pos.getZ(), blockId);
   566→        } catch (Throwable t) {
   567→            Ergenverse.LOGGER.warn("[Ergenverse] CRON-106: Failed to record PLAYER delta "
   568→                    + "for Crystal inheritance at {}: {}", pos, t.getMessage());
   569→        }
   570→    }
